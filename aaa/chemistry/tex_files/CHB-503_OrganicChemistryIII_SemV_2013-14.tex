\noindent\textbf{Question 1.} Answer all parts of the following: \pts{5 $\times$ 2 = 10}
\begin{parts}
  \item Why does benzophenone in isopropanol undergo photoreduction to pinacol upon irradiation at $345\,\text{nm}$?
  \item What happens when $2$-pentanone is irradiated with UV light? (Norrish Type I and Type II reactions).
  \item What is the product when succinaldehyde is treated with ammonia?
  \item What happens when pyridine is heated with $\text{KOH}$ at $320^\circ\text{C}$?
  \item What happens when an unsubstituted amide is treated with sodium hypobromite?
\end{parts}

  \begin{parts}
    \item Benzene $\xrightarrow{\text{i. Na/liq. NH}_3, \text{EtOH}} ?$
    \item Ethyl acetate $\xrightarrow{\text{Na/EtOH}} ?$
    \item Cyclohexanone $\xrightarrow{\text{LDA}, -70^\circ\text{C}, \text{then CH}_3\text{I}} ?$
  \end{parts}
