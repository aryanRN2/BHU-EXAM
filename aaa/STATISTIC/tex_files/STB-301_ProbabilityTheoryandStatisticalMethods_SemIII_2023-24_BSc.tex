\noindent   \textbf{Question 1.}   \textit{(Compulsory)}

\begin{parts}
  \item Distinguish between standard deviation and standard error. If $X_i \sim N(\mu, \sigma^2)$ for $i = 1,2,\dots,n$, find the standard error of the sample mean $ar{X}$ and the sample variance $S^2$.\pts{4}
  \item What do you mean by sample proportion? Derive the sampling distribution of the sample proportion and obtain its standard error.\pts{10}
\end{parts}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item What are the different criteria of a good estimator? Explain them briefly.\pts{6}
  \item If $X_1, X_2, \dots, X_n$ is a random sample from a normal population $N(\mu, \sigma^2)$, show that the sample variance $S^2 = \frac{1}{n-1}\sum_{i=1}^n (X_i - ar{X})^2$ is an unbiased and consistent estimator for the parameter $\sigma^2$.\pts{8}
\end{parts}

\noindent   \textbf{Question 3.}

\begin{parts}
  \item What do you mean by simple and composite hypothesis? A random sample of size $n=1$ is taken from an exponential distribution with parameter $\lambda$ (pdf $f(x) = \lambda e^{-\lambda x}, x > 0$). If $x \geq 1$ is the critical region for testing the hypothesis $H_0: \lambda = 2$ against $H_1: \lambda = 1$, find the type-I error, type-II error and the power of the test.\pts{8}
  \item Define $\chi^2$ (chi-square) distribution, derive the expression of its probability density function, and show that the skewness of the $\chi^2$ distribution with $n$ degrees of freedom is $\sqrt{8/n}$.\pts{6}
\end{parts}
