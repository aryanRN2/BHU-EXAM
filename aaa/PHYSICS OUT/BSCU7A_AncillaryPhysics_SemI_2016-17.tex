\noindent\textbf{Question 1.} Answer any \textbf{six} of the following: \pts{6 $\times$ 3 = 18}
\begin{parts}
  \item What do you understand by streamline flow of a liquid?
  \item What is the difference between stress and strain?
  \item What should be done to reduce the surface tension of water?
  \item Define the entropy of a system.
  \item Give one example of the first law of thermodynamics.
  \item What are the advantages and disadvantages of $\text{CO}_2$ gas in the environment?
  \item How is the image of an extended source formed by a convex lens?
  \item What do you understand by the dispersion of light?
\end{parts}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item What do you mean by Hooke's law? Describe Young's modulus of elasticity. \pts{7}
  \item Discuss Bernoulli's theorem and its applications. \pts{6}
\end{parts}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item Define the surface tension of a liquid. Describe in detail the capillary action of a tube. \pts{7}
  \item What is Stokes' law? Discuss the terminal velocity of a sphere falling in a fluid. \pts{6}
\end{parts}

\medskip\noindent\textbf{Question 4.}
\begin{parts}
  \item Discuss the second law of thermodynamics and its importance. \pts{7}
  \item Discuss the phenomenon of diffusion of gases. On what factors does it depend? \pts{6}
\end{parts}

\medskip\noindent\textbf{Question 5.}
\begin{parts}
  \item Define Helmholtz and Gibbs free energies with some examples. \pts{7}
  \item Discuss Fermat's principle of least time. \pts{6}
\end{parts}

\medskip\noindent\textbf{Question 6.}
\begin{parts}
  \item Discuss Young's double-slit experiment and the conditions for maximum and minimum intensity. \pts{7}
  \item What is the difference between Fraunhofer and Fresnel diffraction? \pts{6}
\end{parts}

\medskip\noindent\textbf{Question 7.}
\begin{parts}
