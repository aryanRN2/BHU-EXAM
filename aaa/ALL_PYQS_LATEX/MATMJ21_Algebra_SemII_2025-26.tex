\noindent   \textbf{Question 1.} Short Answer Questions (Compulsory):\pts{23}

\begin{parts}
  \item Express the numbers $-3$ and $5i$ as complex numbers of the form $a + bi$.\pts{2}
  \item Find two pairs of real numbers $x$ and $y$ for which $(x + iy)^2 = -7 + 24i$.\pts{3}
  \item Show that the primitive cube roots of unity are $\omega$ and $\omega^2$.\pts{3}
  \item Show that $x^n - 1 = (x - 1)(x^{n-1} + x^{n-2} + \dots + x + 1)$.\pts{3}
  \item Divide $x^3 + 6x^2 + 10x - 1$ by $(x - i\sqrt{2})$.\pts{3}
  \item If a polynomial $a_0 x^n + a_1 x^{n-1} + \dots + a_n = 0$ has more than $n$ distinct roots, then \underline{\hspace{3cm}} (fill in the blank).\pts{3}
  \item How many double roots does the equation $x^4 - 6x^3 + 13x^2 - 12x + 4 = 0$ have? Justify your answer.\pts{3}
  \item Write the multiplication (Cayley) table for Klein's four-group $V_4$. Explain your answer.\pts{3}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}




\noindent   \textbf{Question 2.} Theory of Equations:\pts{12}

\begin{parts}
  \item Find a quadratic equation whose roots are the squares of the roots of $x^2 + px + q = 0$.\pts{6}
  \item Solve the equation $x^4 - 2x^3 - 21x^2 + 22x + 40 = 0$ whose roots are in arithmetic progression.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}




\noindent   \textbf{Question 3.} Roots of Polynomials:\pts{12}

\begin{parts}
  \item If an algebraic equation with real coefficients has the root $a + ib$ where $a$ and $b$ are real and $b 
eq 0$, then show that it also has the root $a - ib$.\pts{6}
  \item Find all the integral roots of the equation $f(x) = x^4 - 9x^3 + 24x^2 - 23x + 15 = 0$ by Newton's method.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}




\noindent   \textbf{Question 4.} Matrix Groups \& Modulo Arithmetic:\pts{12}

\begin{parts}
  \item Let $G$ be the set of all $2  \times 2$ matrices $
\begin{pmatrix} a & b \\ c & d \end{pmatrix}$ where $a, b, c, d$ are integers modulo $p$, $p$ being a prime number such that $ad - bc 
eq 0$. Taking matrix multiplication modulo $p$ as the operation in $G$, check whether $G$ is a finite group. Justify your answer.\pts{8}
  \item In part (i) of this question, if the group is finite, determine the order of $G$ for $p = 3$.\pts{4}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}




\noindent   \textbf{Question 5.} Group Properties \& Unique Solutions:\pts{12}

\begin{parts}
  \item If $G$ is a group, then show that: (a) every $a \in G$ has a unique inverse in $G$, and (b) for all $a, b \in G$, $(ab)^{-1} = b^{-1} a^{-1}$.\pts{5}
  \item Let $G$ be a group and $a, b \in G$. Show that the equations $ax = b$ and $ya = b$ have unique solutions for $x$ and $y$ in $G$.\pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}




\noindent   \textbf{Question 6.} Cyclic Groups \& Subgroups:\pts{12}

\begin{parts}
  \item Let $G$ be a finite group and $|G| = p$, a prime number. Show that $G$ is a cyclic group.\pts{6}
  \item In a group $G$ with subgroups $H$ and $K$, define $HK = \{hk \mid h \in H  \text{ and } k \in K\}$. Check whether $HK$ is a subgroup of $G$. Justify your answer.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}




\noindent   \textbf{Question 7.} Symmetry Groups \& Alternating Groups:\pts{12}

\begin{parts}
  \item Determine the symmetry group of a regular pentagon.\pts{7}
  \item For $n \ge 3$, check whether $A_n$ is a subgroup of $S_n$. Is it a normal subgroup? Justify your answer.\pts{5}
\end{parts}
