\noindent   \textbf{Question 1.}

\begin{parts}
  \item Define a convergent sequence of real numbers. Give an example. Show that $\{1/n^2\}_{n \in \mathbb{N}}$ converges to 0.\pts{2.5}
  \item Define a Cauchy sequence of real numbers. Show that every Cauchy sequence of real numbers is bounded.\pts{2.5}
\end{parts}

\bigskip



\noindent   \textbf{Question 2.}

\begin{parts}
  \item Define a monotonic sequence of real numbers. Give an example.\pts{2}
  \item Let $\{a_n\}$ be a sequence of real numbers which is monotonically increasing and bounded above. Show that $\lim_{n  o \infty} a_n$ exists.\pts{3}
\end{parts}

\bigskip



\noindent   \textbf{Question 3.}

\begin{parts}
  \item Define a convergent series of real numbers. Give an example of a series which is not convergent and justify it.\pts{2}
  \item Test the convergence of the series $\sum_{n=1}^{\infty} \left( \sqrt{n^4 + 1} - \sqrt{n^4 - 2} \right)$.\pts{3}
\end{parts}

\bigskip



\noindent   \textbf{Question 4.}

\begin{parts}
  \item Let $I$ be an interval in $\mathbb{R}$ and let $c$ be a limit point of $I$. Give the $\varepsilon-\delta$ definition of limit of a function $f : I  o \mathbb{R}$ as $x  o c$.\pts{2.5}
  \item Define $f(x) = \frac{e^{1/x} - e^{-1/x}}{e^{1/x} + e^{-1/x}}$ for $x 
eq 0$. Does $\lim_{x  o 0} f(x)$ exist?\pts{2.5}
\end{parts}
