\noindent\textbf{Question 1.} Give the definition of curl of a vector function. If $\vec{A}$ and $\vec{B}$ are two vector functions, then prove that
\[
  \text{curl}(\vec{A} \times \vec{B}) = (\vec{B} \cdot \nabla)\vec{A} - \vec{B}(\nabla \cdot \vec{A}) - (\vec{A} \cdot \nabla)\vec{B} + \vec{A}(\nabla \cdot \vec{B}).
\]\pts{5}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 2.} Give the definition of gradient of a scalar function. Find the equation of tangent plane to the surface $4x^2 z + yz^2 + xy^2 = 2$ at the point $(1, 2, -1)$.\pts{5}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 3.} Prove that
\begin{enumerate}[label=(\roman*)]
  \item $\text{curl}(\text{grad }\phi) = 0$,
  \item $\text{div}(\text{curl }\vec{A}) = 0$,
\end{enumerate}
where $\phi$ is a scalar function and $\vec{A}$ is a vector function.\pts{5}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 4.} Evaluate
\[
  \int_{c} (2xz^2 \, dx + 3z^2 \, dy - 3x^2 y \, dz),
\]
where $c$ is the curve given by $x = t^3$, $y = 2t$, $z = t^2$, from $t = 0$ to $t = 2$.\pts{5}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 5.} Evaluate
\[
  \iint_{S} \vec{F} \cdot \hat{n} \, ds,
\]
where $\vec{F} = 2y\hat{i} - z\hat{j} + x^2\hat{k}$ and $S$ is the surface of the cylinder $y^2 = 8x$ in the first octant bounded by the planes $y = 4$ and $z = 6$.\pts{5}
