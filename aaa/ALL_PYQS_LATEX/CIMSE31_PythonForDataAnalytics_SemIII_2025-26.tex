\noindent\textbf{Question 1.} Short Answer Questions (Compulsory):\pts{14}
\begin{parts}
  \item Which keyword is used to define a function in Python?\pts{2}
  \item What will be the output produced by the following Python code?\pts{2}
\begin{verbatim}
a = 5
b = -3
c = 25
d = -10

print(a + b + c > a + c - b * d)
print(str(a + b + c > a + c - b * d) == 'True')
print(len(str(a + b + c > a + c - b * d)) == len(str(bool(1))))
\end{verbatim}
  \item Which data structure stores data in key-value pairs in Python?\pts{2}
  \item What symbol is used for single-line comments in Python?\pts{2}
  \item What will be the output produced by the following Python code?\pts{2}
\begin{verbatim}
a = 3 + 5 / 8
b = int(3 + 5 / 8)
c = 3 + float(5 / 8)
d = 3 + float(5) / 8
e = 3 + 5.0 / 8
f = int(3 + 5 / 8.0)

print(a, b, c, d, e, f)
\end{verbatim}
  \item What type of plot is used to show the relationship between two numerical variables in Data Analytics?\pts{2}
  \item What is the basic 1D data structure unit of a Pandas DataFrame column?\pts{2}
\end{parts}
