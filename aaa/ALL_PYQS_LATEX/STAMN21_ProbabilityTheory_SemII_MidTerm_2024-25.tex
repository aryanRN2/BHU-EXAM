\section*{Section- A}



\noindent   \textit{Attempt all questions. Each question carries 2 marks.}
\bigskip


\begin{enumerate}[label=   \textbf{1. }]
  \item
  
\begin{parts}
    \item If A, B and C are any three mutually exclusive and exhaustive events of a random experiment such that $P(B) = \frac{3}{2} P(A)$ and $P(C) = \frac{1}{2} P(B)$. Then find the value of $P(A)$.\pts{2}

    \item If A and B are disjoint events of a random experiment, then prove that $P(A) \le P(ar{B})$.\pts{2}

    \item State Boole's inequality.\pts{2}

    \item Define conditional probability and distinguish it from the unconditional probability.\pts{2}

    \item If A and B are any two mutually exclusive events with positive probabilities, then prove that they are always dependent events.\pts{2}
  \end{parts}
\end{enumerate}

\section*{Section- B}



\noindent   \textit{Attempt any two questions out of the following four questions. Each question carries 5 marks.}
\bigskip


\begin{enumerate}[label=   \textbf{2. }]
  \item
  
\begin{parts}
    \item Prove that the result $P(A \cap B) = P(A) P(B)$ is a necessary and sufficient condition for any two events A and B of a random experiment with positive probabilities to be their independence.\pts{5}

    \item Define pairwise independence and mutual independence of any n events of a random experiment. Also prove that for mutual independence of n events $A_1, A_2, \dots, A_n$, we have to check $2^n - n - 1$ conditions.\pts{5}

    \item State Bayes theorem. Further, the probability that a doctor H will diagnose a disease D incorrectly is 40\%. The chance that a patient will die by his treatment after correct diagnosis is 40\% and the chance of death by wrong diagnosis is 70\%. A patient of doctor H, who had disease D, died. What is the probability that his disease was diagnosed correctly.\pts{5}

    \item Define probability density function (PDF) of a continuous random variable. Next, a continuous random variable X has the PDF $f(x) = 3x^2; 0 \le x \le 1$. Find the constants a and b such that $P(X \le a) = P(X > a)$ and $P(X > b) = 0.05$.\pts{5}
  \end{parts}
\end{enumerate}
