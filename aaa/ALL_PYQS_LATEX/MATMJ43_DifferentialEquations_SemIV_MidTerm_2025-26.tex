\noindent\textbf{Question 1.} Solve Clairaut's differential equation $y = px + f(p)$ where $p = \frac{dy}{dx}$, and obtain its general and singular solutions.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 2.} Solve the second-order linear ODE $\frac{d^2 y}{dx^2} - 4\frac{dy}{dx} + 4y = x^2 e^{2x}$.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 3.} Solve by the Method of Variation of Parameters: $\frac{d^2 y}{dx^2} + y = \sec x$.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 4.} Solve Cauchy-Euler homogeneous linear differential equation $x^2 \frac{d^2 y}{dx^2} - 3x \frac{dy}{dx} + 4y = x^2 \ln x$.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 5.} Find the orthogonal trajectories of the family of cardioids $r = a(1 + \cos\theta)$.\pts{4}
