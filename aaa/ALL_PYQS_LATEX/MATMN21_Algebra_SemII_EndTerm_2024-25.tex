\noindent   \textbf{Question 1.}   \textit{(Compulsory, 14 Marks Total)}

\begin{parts}
  \item Express $-1 - \sqrt{3}i$ in modulus-amplitude form.\pts{2}
  \item If $S = \left\{ \frac{1}{n} \;\middle|\; n \in \mathbb{N} \right\}$, then determine infimum of $S$.\pts{2}
  \item Find all the roots of equation $Z^3 = -8i$, where $Z \in \mathbb{C}$ and $i = \sqrt{-1}$.\pts{2}
  \item State the fundamental theorem of Algebra. Determine all the roots of polynomial $p(x) = x^3 - 5x^2 - 16x + 80$.\pts{1+1=2}
  \item Define a relation $R$ on $\mathbb{Z} - \{1\}$ by declaring that $xRy$ if $x$ and $y$ have a common factor other than $1$. Check whether $R$ is an equivalence relation.\pts{2}
  \item Let $G$ be a cyclic group and $a (
eq e)$ be an element of $G$. If $a$ generates $G$, then show that order of $G$ is same as order of $a$.\pts{2}
  \item Let $n \in \mathbb{Z}$ be a positive integer. Recall the congruence relation `$\equiv$' defined on $\mathbb{Z}$. If $a \equiv b \pmod n$ and $c \equiv d \pmod n$ for some $a, b, c, d$ in $\mathbb{Z}$, then verify whether $ac \equiv bd \pmod n$.\pts{2}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 2.}

\begin{parts}
  \item Use Euclidean algorithm to show that: if $a, b \in \mathbb{Z} - \{0\}$, then there exist $x, y \in \mathbb{Z}$ such that $ax + by = \gcd\{a, b\}$.\pts{7}
  \item Solve the following equation in the set of integers: $237x + 81y = 15$.\pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 3.}

\begin{parts}
  \item Let $S$ be the set of all real numbers except $-1$. Define an operation `$*$' on $S$ by $a * b = a + b + ab$. Examine whether $S$ is a group with the operation `$*$'. Is it an Abelian group? Justify.\pts{8}
  \item For $n = 12$ consider $\mathbb{Z}_n$. For each divisor $m$ of $n$, write down an element $a \in \mathbb{Z}_n$ such that order of $a$ is $m$.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 4.}

\begin{parts}
  \item Define subgroup of a group. Prove that a non-empty subset $H$ of a group $G$ is a subgroup if and only if $a, b \in H \implies ab^{-1} \in H$.\pts{7}
  \item Show that a subgroup of a cyclic group is always cyclic.\pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 5.}

\begin{parts}
  \item Define centre of a group and centralizer of an element. Consider the group $Q_8 = \{\pm 1, \pm i, \pm j, \pm k\}$ which satisfies the following multiplication rules:
  \[
    i^2 = j^2 = k^2 = -1, \quad ij = k, \; jk = i, \; ki = j, \quad ji = -k, \; kj = -i, \; ik = -j.
  \]
  Find the centre of $Q_8$ and centralizer of $i$.\pts{6}
  \item Does the group of symmetries of a square contain an element of order six? Justify your answer.\pts{8}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 6.}

\begin{parts}
  \item Determine the roots of the polynomial $x^3 + 10x + 5$ in $\mathbb{Z}$. What can you say about its roots in $\mathbb{Q}$?\pts{7}
  \item Find all the roots of polynomial $x^6 - 1 = f(x)$. Can you use Newton's method to find some roots of this polynomial? Justify your answer.\pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 7.}

\begin{parts}
  \item Define a binary operation on the set of even integers which is different from addition, subtraction or multiplication.\pts{2}
  \item Find a solution of the equation $ax = b$ in $S_3$, where
  \[
    a = 
\begin{pmatrix} 1 & 2 & 3 \\ 3 & 2 & 1 \end{pmatrix}, \quad b = 
\begin{pmatrix} 1 & 2 & 3 \\ 1 & 3 & 2 \end{pmatrix}.
  \]\pts{3}
  \item Give an example of a group which is not cyclic, but its every proper subgroup is cyclic.\pts{2}
  \item Let $G$ be a group and $a, b \in G$. Prove the following:
  
\begin{parts}
    \item $(ab)^{-1} = b^{-1}a^{-1}$
    \item $(ab)^{-1} = a^{-1}b^{-1} \iff G  \text{ is Abelian}$.
  \end{parts}\pts{2}
  \item Define order of an element of a group. Let $O(a) = n$, show the following:
  
\begin{parts}
    \item $a^m = e \iff m = nq$, where $q \in \mathbb{Z}$
    \item $O(a^s) \le n$, where $s \in \mathbb{Z}$
    \item If $p$ is prime to $n$, then $O(a^p) = n$.
  \end{parts}\pts{1+1+2=4}
\end{parts}
