\noindent\textbf{Question 2.}
\begin{parts}
  \item Define subgroup. Prove that a non-empty subset $H$ of a group $G$ is a subgroup iff $a b^{-1} \in H$ for all $a, b \in H$.\pts{2.5}
  \item State and prove Lagrange's Theorem for finite groups.\pts{2.5}
\end{parts}

\bigskip
\noindent\textbf{Question 3.}
\begin{parts}

\end{parts}
