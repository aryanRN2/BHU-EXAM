\bigskip

\centerline{   \textbf{SECTION A}}
\smallskip



\noindent   \textit{Note: Objective type questions of 2 marks each, maximum 50 words.\hfill [$5  \times 2 = 10$ Marks]}


\begin{enumerate}[label=   \textbf{1.}]
    \item Answer the following:
    
\begin{parts}
        \item Who is the author of the book `Nationalist Thought and the Colonial World : A Derivative Discourse'?
        \item Distinguish between art and imitation.
        \item Who is the ideal artist according to Sri Aurobindo?
        \item What do you mean by `Trusteeship'?
        \item Who is known as the father of modern aesthetics?
    \end{parts}
\end{enumerate}

\bigskip
\rule{\linewidth}{0.2pt}
\medskip

\centerline{   \textbf{SECTION B}}
\smallskip



\noindent   \textit{Note: Short answer type questions of 250 words each. Each of 10 marks. Attempt any three questions.\hfill [$3  \times 10 = 30$ Marks]}


\begin{enumerate}[label=   \textbf{\arabic*.}]
    \setcounter{enumi}{1}
    \item Discuss Bankimchandra's concept of Hindu Nationalism.
    
    
\begin{center}   \textbf{OR}\end{center}

    In what ways did Gandhi's practices of fasting and simplicity reflect his interpretation of self and nation?

    \item Analyze Deendayal Upadhyay's views on the Western Notion of Individualism versus the Indian concept of self (   \textit{atma}).
    
    
\begin{center}   \textbf{OR}\end{center}

    Examine Gandhi's idea of Nation and Nationalism. How does his approach differ from European models of Nationalism?

    \item Critically evaluate M. N. Roy's concept of Radical Humanism. How does it offer an alternative to both Liberalism and Marxism?
    
    
\begin{center}   \textbf{OR}\end{center}

    Explain Coomaraswamy's views on the spiritual foundations of Indian Art.
\end{enumerate}

\bigskip
\rule{\linewidth}{0.2pt}
\medskip

\centerline{   \textbf{SECTION C}}
\smallskip



\noindent   \textit{Note: Long answer type questions each of 15 marks. Attempt any two questions.\hfill [$2  \times 15 = 30$ Marks]}


\begin{enumerate}[label=   \textbf{\arabic*.}]
    \setcounter{enumi}{4}
    \item Critically evaluate the relevance of `Integral Humanism' in contemporary India.
    
    
\begin{center}   \textbf{OR}\end{center}

    Critically examine Rabindranath Tagore's idea of the self in relation to the universal human spirit.

    \item Critically analyze Ramanujan's interpretation of Indian folklore and oral traditions. How does he connect them with the idea of Indian cultural identity?
    
    
\begin{center}   \textbf{OR}\end{center}

    Discuss Aurobindo's vision of India as a spiritual nation. How does he define the essence of Indian civilization?
\end{enumerate}
