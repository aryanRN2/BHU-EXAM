\noindent\textbf{Question 1.} \textit{(Compulsory)}
\begin{parts}
  \item Give the definition of a convergent sequence of real numbers. Prove that the sequence $\{1/n^2\}_{n \in \mathbb{N}}$ converges to 0.\pts{3}
  \item Does $\lim_{x \to 0} \frac{x}{|x|}$ exist? Justify your answer.\pts{3}
  \item Let $f : \mathbb{R} \to \mathbb{R}$ be defined by $f(x) = x \sin(1/x)$ if $x \neq 0$ and $f(0) = p \in \mathbb{R}$. Find the value of $p$ for which $f$ is continuous at $x = 0$.\pts{2}
  \item Let $f : (a,b) \to \mathbb{R}$ be a function such that $f'(c)$ exists for some $c \in (a,b)$. Show that $f$ is continuous at $c$.\pts{2}
  \item Evaluate $\lim_{n \to \infty} \left[ \frac{1}{n+1} + \frac{1}{n+2} + \dots + \frac{1}{n+n} \right]$.\pts{2}
  \item Define point of inflexion of a curve. Find the points of inflexion of the curve $y^2 = x(1+x^2)$.\pts{2}
\end{parts}

\bigskip
\noindent\textbf{Question 2.}
\begin{parts}
  \item Let $\{a_n\}_{n \in \mathbb{N}}$ be a sequence in $\mathbb{R}$ such that $\{a_n\}$ converges to $L \in \mathbb{R}$. Show that $\{a_n\}$ is a Cauchy sequence.\pts{5}
  \item Define a sequence $\{a_n\}$ by $a_n = 1 + \frac{1}{2} + \dots + \frac{1}{n}$ for all $n \in \mathbb{N}$. Show that $\{a_n\}$ is not Cauchy.\pts{5}
  \item State Cauchy's first theorem on limits. Show that $\lim_{n \to \infty} \frac{1}{n}\left(1 + \frac{1}{2} + \dots + \frac{1}{n}\right) = 0$.\pts{4}
\end{parts}

\bigskip
\noindent\textbf{Question 3.}
\begin{parts}
  \item State and prove D'Alembert's ratio test for series of real numbers.\pts{7}
  \item Test the convergence of the series $\sum_{n=1}^{\infty} \frac{n!}{n^n}$.\pts{7}
\end{parts}

\bigskip
\noindent\textbf{Question 4.}
\begin{parts}
  \item Let $I$ be an interval of $\mathbb{R}$ and let $p$ be a limit point of $I$. Let $f,g : I \to \mathbb{R}$ be functions such that $\lim_{x \to p} f(x) = L$ and $\lim_{x \to p} g(x) = M$. Show that $\lim_{x \to p} f(x)g(x) = LM$.\pts{7}
  \item Let $f : [a,b] \to \mathbb{R}$ be a continuous function. Show that there exist $u,v \in [a,b]$ such that $f(u) = \inf_{x \in [a,b]} f(x)$ and $f(v) = \sup_{x \in [a,b]} f(x)$.\pts{7}
\end{parts}

\bigskip
\noindent\textbf{Question 5.}
\begin{parts}
  \item State and prove Lagrange's Mean Value Theorem.\pts{7}
  \item Define local minimum point and local maximum point of a real-valued function. Let $f(x) = \sin x (1 + \cos x), x \in \mathbb{R}$. Find $f''(x)$ and determine the maximum value of $f$.\pts{7}
\end{parts}

\bigskip
\noindent\textbf{Question 6.}
\begin{parts}
  \item State and prove Leibniz's theorem for the $n$-th derivative of the product of two differentiable functions.\pts{7}
  \item Expand the function $f(x) = \log(1+x), x \in (-1,1)$ in powers of $x$.\pts{7}
\end{parts}

\bigskip
\noindent\textbf{Question 7.}
\begin{parts}
  \item Find all asymptotes of the curve $x^3 + 2x^2 y + xy^2 - y + 1 = 0$.\pts{7}
  \item Trace the curve $x^3 + y^3 = 3axy, a > 0$.\pts{7}
\end{parts}
