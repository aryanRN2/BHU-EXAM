\section*{Question 1 (Compulsory)}



\noindent   \textbf{Choose the correct option from the following:} \hfill [$10  \times 2 = 20$]


\begin{enumerate}[label=(\alph*)]
    \item What is Android?
    
\begin{enumerate}[label=(\roman*)]
        \item A mobile device \quad (ii) An operating system \quad (iii) A web browser \quad (iv) A programming language
    \end{enumerate}
    
    \item Which file is the entry point for an Android application configuration?
    
\begin{enumerate}[label=(\roman*)]
        \item   \texttt{MainActivity.kt} \quad (ii)   \texttt{AndroidManifest.xml} \quad (iii)   \texttt{strings.xml} \quad (iv)   \texttt{build.gradle}
    \end{enumerate}

    \item What does APK stand for?
    
\begin{enumerate}[label=(\roman*)]
        \item Android Phone Kit \quad (ii) Android Package Kit \quad (iii) Android Primary Key \quad (iv) Application Package Key
    \end{enumerate}

    \item Which language is officially recommended by Google for Android development since 2019?
    
\begin{enumerate}[label=(\roman*)]
        \item Kotlin \quad (ii) C\# \quad (iii) JavaScript \quad (iv) Ruby
    \end{enumerate}

    \item What is the purpose of the   \texttt{R.java} file?
    
\begin{enumerate}[label=(\roman*)]
        \item To store app settings \quad (ii) To manage database \quad (iii) To link XML resources with Java/Kotlin code \quad (iv) To debug code
    \end{enumerate}

    \item Which component is used to display a UI screen in Android?
    
\begin{enumerate}[label=(\roman*)]
        \item Broadcast Receiver \quad (ii) Service \quad (iii) Activity \quad (iv) Content Provider
    \end{enumerate}

    \item Which lifecycle method is called when an Activity is first created?
    
\begin{enumerate}[label=(\roman*)]
        \item   \texttt{onStart()} \quad (ii)   \texttt{onCreate()} \quad (iii)   \texttt{onResume()} \quad (iv)   \texttt{onRestart()}
    \end{enumerate}

    \item Which layout arranges elements linearly in a single column or row?
    
\begin{enumerate}[label=(\roman*)]
        \item   \texttt{ConstraintLayout} \quad (ii)   \texttt{GridLayout} \quad (iii)   \texttt{FrameLayout} \quad (iv)   \texttt{LinearLayout}
    \end{enumerate}

    \item Which of the following is used for local relational data storage in Android?
    
\begin{enumerate}[label=(\roman*)]
        \item Firebase \quad (ii) MySQL \quad (iii) SQLite \quad (iv) PostgreSQL
    \end{enumerate}

    \item What component allows sharing of data between different applications in Android?
    
\begin{enumerate}[label=(\roman*)]
        \item Content Provider \quad (ii) Intent \quad (iii) Service \quad (iv) Fragment
    \end{enumerate}
\end{enumerate}
