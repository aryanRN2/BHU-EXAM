\bigskip




\noindent   \textbf{Question 1.}

\begin{parts}
  \item Define a convergent sequence of real numbers. Give an example of a convergent sequence. Show that
  \[
    \lim_{n  o \infty} (n)^{1/n} = 1.
  \]\pts{2.5}
  \item Define a Cauchy sequence of real numbers. Show that every Cauchy sequence of real numbers is bounded.\pts{2.5}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 2.}

\begin{parts}
  \item Define a monotonic sequence of real numbers. Give an example of a monotonic sequence of real numbers.\pts{2}
  \item Let $\{a_n\}$ be a sequence of real numbers such that it is monotonically increasing and it has an upper bound. Show that $\lim_{n  o \infty} a_n = \ell$ exists. Find the value of $\ell$.\pts{3}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 3.}

\begin{parts}
  \item Define a convergent series of real numbers. Give an example of a series of real numbers, which is not convergent and justify it.\pts{2}
  \item Test the convergence of the series
  \[
    \sum_{n=1}^{\infty} \left( \sqrt{n^4 + 1} - \sqrt{n^4 - 1} \right).
  \]\pts{3}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 4.}

\begin{parts}
  \item Let $I$ be an interval in $\mathbb{R}$ and let $c$ be a limit point of $I$. Give $\varepsilon$-$\delta$ definition of limit of a function $f : I  o \mathbb{R}$ at $c$ as $x$ approaches to $c$. Give an example in support of $\varepsilon$-$\delta$ definition of limit of a function.\pts{2.5}
  \item Define a function $f : (-\infty, 0) \cup (0, \infty)  o \mathbb{R}$ by
  \[
    f(x) = \frac{e^{1/x} - e^{-1/x}}{e^{1/x} + e^{-1/x}}, \quad x 
eq 0.
  \]
  Does $\lim_{x  o 0} f(x)$ exist? Find $\lim_{x  o 0} f(x)$ if it exists.\pts{2.5}
\end{parts}
