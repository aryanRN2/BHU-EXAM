\noindent\textbf{Question 1.} \textit{(Compulsory)}
\begin{parts}
  \item Define mean, median, and mode. Write down the empirical relationship between them.\pts{3}
  \item Define variance and standard deviation. Show that variance is independent of change of origin but depends on change of scale.\pts{3}
  \item Define raw moments and central moments. Express the first four central moments in terms of raw moments.\pts{3}
  \item State the axiomatic definition of probability.\pts{2.5}
  \item State Bayes' Theorem for conditional probability.\pts{2.5}
\end{parts}
