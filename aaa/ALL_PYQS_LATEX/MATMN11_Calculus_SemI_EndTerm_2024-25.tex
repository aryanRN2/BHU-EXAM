\noindent   \textbf{Question 1.} Short Answer Questions (Compulsory):\pts{14}

\begin{parts}
  \item Give the definition of a convergent sequence of real numbers. Prove that the sequence $\left\{\frac{1}{n^2}\right\}_{n \in \mathbb{N}}$ converges to 0.\pts{3}
  \item Does $\lim_{x  o 0} \sin\left(\frac{1}{x}\right)$ exist? Justify your answer.\pts{3}
  \item Let $f \colon \mathbb{R}  o \mathbb{R}$ be the function defined by
  \[
    f(x) = 
\begin{cases} x \sin\left(\frac{1}{x}\right), &  \text{if } x 
eq 0, \\ p, &  \text{if } x = 0,  \text{ where } p \in \mathbb{R}. \end{cases}
  \]
  Find the value of $p$ for which the function $f$ is continuous at $x = 0$.\pts{2}
  \item Let $f \colon (a, b)  o \mathbb{R}$ be a function such that $f'(c)$ exists for some $c \in (a, b)$. Show that $f$ is continuous at $c$.\pts{2}
  \item Evaluate $\lim_{n  o \infty} \left[ \frac{1}{1+n} + \frac{1}{2+n} + \dots + \frac{1}{n+n} \right]$.\pts{2}
  \item Define point of inflexion of a curve. Find the points of inflexion of the curve $y^2 = x(1+x^2)$.\pts{2}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 2.}

\begin{parts}
  \item Let $\{a_n\}_{n \in \mathbb{N}}$ be a sequence in $\mathbb{R}$ such that $\{a_n\}$ converges to $L \in \mathbb{R}$. Show that $\{a_n\}_{n \in \mathbb{N}}$ is a Cauchy sequence.\pts{5}
  \item Define a sequence $\{a_n\}_{n \in \mathbb{N}}$ by $a_n = 1 + \frac{1}{2} + \dots + \frac{1}{n}$ for all $n \in \mathbb{N}$. Show that $\{a_n\}$ is not a Cauchy sequence.\pts{5}
  \item State Cauchy's first theorem on limits. Show that $\lim_{n  o \infty} \frac{1}{n} \left( 1 + \frac{1}{2^{1/2}} + \dots + \frac{1}{n^{1/n}} \right) = 1$ by using Cauchy's first limit theorem.\pts{4}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 3.}

\begin{parts}
  \item State and prove D'Alembert's ratio test for series of real numbers.\pts{7}
  \item Test the convergence of the series $\sum_{n=1}^\infty \frac{n! x^n}{(n+1)^n}$.\pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 4.}

\begin{parts}
  \item Let $f \colon \mathbb{R}  o \mathbb{R}$ be a function defined by
  \[
    f(x) = 
\begin{cases} e^{-1/x^2} \sin\left(\frac{1}{x}\right), &  \text{if } x 
eq 0, \\ 0, &  \text{if } x = 0. \end{cases}
  \]
  Test the differentiability of $f$ at $x = 0$.\pts{7}
  \item Let $f$ and $g$ be real-valued functions defined on the interval $(a, b)$ such that $f$ and $g$ are differentiable at $c \in (a, b)$. Show that:
  
\begin{parts}
    \item $f + g$ is differentiable at $c$ and $(f+g)'(c) = f'(c) + g'(c)$.
    \item $fg$ is differentiable at $c$ and $(fg)'(c) = f'(c)g(c) + f(c)g'(c)$.
  \end{parts}\pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip
