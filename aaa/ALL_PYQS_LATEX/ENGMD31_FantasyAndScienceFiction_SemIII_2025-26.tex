\section*{SECTION -- A (Long Answer Type Questions)}
\noindent \textbf{Answer the following questions in about 400 words each:} \hfill [$3 \times 12 = 36$ Marks]

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item Describe the historical evolution of Fantasy and Science Fiction as distinct literary genres.
    \begin{center}\textbf{OR}\end{center}
    Analyse the impact of time travel and artificial intelligence on the imaginative scope of science fiction narratives.

    \item Devakinandan Khatri's \textit{Chandrakanta} is a foundational work in Indian fantasy fiction. Elaborate with examples from the text.
    \begin{center}\textbf{OR}\end{center}
    How does Angela Carter's \textit{The Snow Child} reinterpret the traditional fairy tale from a feminist perspective?

    \item Evaluate Arthur C. Clarke's \textit{2001: A Space Odyssey} as a milestone in Science Fiction literature.
    \begin{center}\textbf{OR}\end{center}
    Examine Ray Bradbury's \textit{Mars Is Heaven!} as a Science Fiction narrative.
\end{enumerate}
