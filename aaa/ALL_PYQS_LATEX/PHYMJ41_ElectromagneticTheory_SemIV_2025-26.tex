\noindent   \textbf{Question 2.}

\begin{parts}
  \item What are Lorentz transformations? Show that the combination $x^2 - c^2 t^2$ remains invariant under Lorentz transformations.\pts{$7\frac{1}{2}$}
  \item Show that the wave equation with velocity $c$ does not change under Lorentz transformations.\pts{5}
\end{parts}

\bigskip



\noindent   \textbf{Question 3.}

\begin{parts}
  \item Consider a reference frame $R_2$ moving with velocity $v_2$ with respect to reference frame $R_1$, which in turn moves with velocity $v_1$ with respect to reference frame $R_0$. What is the velocity of $R_2$ with respect to $R_0$?\pts{5}
  \item A stick of length $34 \text{ cm}$ (in its rest frame) is moving at speed $v = \frac{15}{17}c$ with respect to observer A. What will be the length of the stick according to observer A?\pts{5}
  \item A graviton (the particle carrying gravitational force) has vanishing rest mass. What would be its velocity with respect to the ground?\pts{$2\frac{1}{2}$}
\end{parts}

\bigskip



\noindent   \textbf{Question 4.}

\begin{parts}
  \item Write down Maxwell's equations in vacuum. Which of these equations satisfy vector identities?\pts{4}
  \item Show that the electric field obeys a wave equation in vacuum.\pts{$4\frac{1}{2}$}
  \item Write down Maxwell's equations in a medium. How do they simplify for linear media? (Derivation not required).\pts{4}
\end{parts}

\bigskip



\noindent   \textbf{Question 5.} Derive the expressions for the reflection and transmission coefficients when an electromagnetic wave is incident obliquely on a boundary separated by two lossless dielectrics for the case when the electric field vector of the electromagnetic wave is perpendicular to the plane of incidence.\pts{$12\frac{1}{2}$}

\bigskip



\noindent   \textbf{Question 6.}

  \item Calculate the intrinsic impedance of free space.\pts{$3\frac{1}{2}$}
\end{parts}
