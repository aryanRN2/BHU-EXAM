\noindent \textit{Instructions: Answer five questions including Question 1 which is compulsory. All symbols have their usual meanings unless specified otherwise.}

\bigskip

\noindent \textbf{Question 1.} Answer any four of the following questions: \pts{4 $\times$ 5 = 20}
\begin{enumerate}[label=(\alph*)]
  \item Define the gradient of a scalar function. Find the gradient of the function $f(x, y, z) = x^2 y^3 z^4$.
  \item What is the magnetic field outside an infinite solenoid with current carrying wires with current density $I$ wrapping it?
  \item For the vector $\vec{v} = (2xz + 3y^2)\hat{a}_y + (4yz^2)\hat{a}_z$, calculate the line integral for the closed square path in $y-z$ plane.
  \item What are Lorentz and Coulomb gauges?
  \item Find the skin depth of copper whose conductivity $\sigma = 5.8 \times 10^7\text{ mho/m}$, $\mu = \mu_0$ at frequency $10^8\text{ Hz}$.
  \item Find the electric field at a distance $d$ from an infinite sheet with uniform surface charge density $\sigma$.
\end{enumerate}

\bigskip

\noindent \textbf{Question 2.}
\begin{enumerate}[label=(\alph*)]
  \item What are the Lorentz transformations? Show that the combination $x^2 - c^2 t^2$ does not change under Lorentz transformations. \pts{7$\frac{1}{2}$}
  \item Show that wave equation with velocity $c$, does not change under Lorentz transformations. \pts{5}
\end{enumerate}

\bigskip

\noindent \textbf{Question 3.}
\begin{enumerate}[label=(\alph*)]
  \item Consider a reference frame $R_2$ moving in a velocity $v_2$ with respect to a reference frame $R_1$, which in turn moves in a velocity $v_1$ with respect to another reference frame $R_0$. What is the velocity of $R_2$ with respect to $R_0$? \pts{5}
  \item A stick of length $34\text{ cm}$ (in rest frame) is moving at the speed $v = \frac{15}{17}c$ with respect to observer A. What will be the length of the stick according to A? \pts{5}
  \item A graviton (the particle carrying gravitational force) has vanishing rest mass. What would be its velocity with respect to ground? \pts{2$\frac{1}{2}$}
\end{enumerate}

\bigskip

\noindent \textbf{Question 4.}
\begin{enumerate}[label=(\alph*)]
  \item Write down the Maxwell's equations in vacuum. Which of these equations satisfy vector identities? \pts{4}
  \item Show that the electric field obeys a wave equation. \pts{4$\frac{1}{2}$}
  \item Write down Maxwell's equations in a medium. How do they simplify for a linear media? (derivation not required) \pts{4}
\end{enumerate}

\bigskip

\noindent \textbf{Question 5.} Derive the expressions for the reflection and transmission coefficients when an electromagnetic wave incident obliquely on a boundary separated by two lossless dielectrics for the case when electromagnetic wave electric field is perpendicular to the plane of incidence. \pts{12$\frac{1}{2}$}

\bigskip

\noindent \textbf{Question 6.}
\begin{enumerate}[label=(\alph*)]
  \item Discuss the propagation of electromagnetic wave in conducting media and derive attenuation and phase constants. \pts{9}
  \item Calculate the intrinsic impedance of free space. \pts{3$\frac{1}{2}$}
\end{enumerate}

\bigskip

\noindent \textbf{Question 7.} State the Stoke's theorem for a vector field. Suppose vector field $\vec{v} = xy \hat{a}_x + 2yz \hat{a}_y + 3zx \hat{a}_z$. Check the Stoke's theorem using a triangular surface with vertices $(0,0,0)$, $(0,2,0)$, $(0,0,2)$ in $y-z$ plane. \pts{12$\frac{1}{2}$}
