\noindent   \textbf{Question 1}

\begin{parts}
  \item State and prove Leibniz's Theorem for finding the $n$-th derivative of the product of two functions.\pts{6}
  \item If $y = (x^2 - 1)^n$, prove that $(x^2 - 1) y_{n+2} + 2x y_{n+1} - n(n+1) y_n = 0$.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 2}

\begin{parts}
  \item Find all asymptotes of the curve $x^3 + 2x^2 y - x y^2 - 2y^3 + 4x^2 - y^2 + 2x + 1 = 0$.\pts{6}
  \item Derive the expression for radius of curvature $\rho$ in polar coordinates $r = f( \theta)$.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 3}

\begin{parts}
  \item Define envelope and evolute of a family of curves. Find the envelope of the family of straight lines $\frac{x}{a} + \frac{y}{b} = 1$, where the parameters $a$ and $b$ are connected by $a + b = c$ ($c$ is constant).\pts{6}
  \item Find the coordinates of the center of curvature at any point $(x, y)$ on the parabola $y^2 = 4ax$.\pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 4}
Trace the curve $y^2(a - x) = x^3$ (Cissoid of Diocles) completely, giving details of symmetry, origin, asymptotes, tangents at origin, and region of existence.\pts{12}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 5}

\begin{parts}
  \item Find the points of inflexion on the curve $y = \frac{c^2 x}{x^2 + a^2}$.\pts{6}
  \item Determine the concavity and convexity of the curve $y = x^4 - 6x^3 + 12x^2 - 8x + 1$.\pts{6}
\end{parts}
