\noindent\textbf{Question 1.} Define double limit and repeated limits for $f(x, y)$ at $(0, 0)$. Show that for $f(x, y) = \frac{xy}{x^2 + y^2}$, repeated limits exist and are equal, but double limit does not exist.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 2.} State and prove Schwarz's Theorem on the equality of mixed partial derivatives $f_{xy}(a, b)$ and $f_{yx}(a, b)$.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 3.} State and prove Euler's Theorem on homogeneous functions of two variables $x$ and $y$ of degree $n$.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 4.} Discuss the maxima and minima of $f(x, y) = x^3 + y^3 - 3axy$.\pts{4}

\medskip\rule{\linewidth}{0.2pt}\medskip

\noindent\textbf{Question 5.} State Lagrange's Method of Undetermined Multipliers and find the extreme values of $f(x, y, z) = x^2 + y^2 + z^2$ subject to $x + y + z = 3$.\pts{4}
