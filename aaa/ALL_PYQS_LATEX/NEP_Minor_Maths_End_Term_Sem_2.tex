\noindent   \textbf{Question 1.}   \textit{(Compulsory)}

\begin{parts}
  \item Define cyclic group and give an example.\pts{3}
  \item Prove that every subgroup of a cyclic group is cyclic.\pts{3}
  \item Define permutation group and symmetric group $S_n$.\pts{3}
  \item State Cayley's Theorem for finite groups.\pts{2.5}
  \item Define ring, integral domain, and field.\pts{2.5}
\end{parts}

\bigskip



\noindent   \textbf{Question 2.}

\begin{parts}
  \item State and prove Lagrange's Theorem for finite groups.\pts{7}
  \item Find all subgroups of the cyclic group $(\mathbb{Z}_{12}, +_{12})$.\pts{7}
\end{parts}

\bigskip



\noindent   \textbf{Question 3.}

\begin{parts}
  \item Define normal subgroup and quotient group.\pts{7}
  \item State and prove the First Isomorphism Theorem for groups.\pts{7}
\end{parts}

\bigskip



\noindent   \textbf{Question 4.}

\begin{parts}
  \item Define ideal of a ring. Prove that the intersection of two ideals of a ring $R$ is an ideal of $R$.\pts{7}
  \item Define prime ideal and maximal ideal with suitable examples.\pts{7}
\end{parts}
