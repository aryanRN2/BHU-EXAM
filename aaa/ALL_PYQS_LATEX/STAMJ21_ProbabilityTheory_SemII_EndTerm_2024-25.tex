\noindent   \textbf{Question 1.}

\begin{parts}
  \item A fair die is tossed three times. Let $A_i$ be the event that $i$-th toss turns up with even number, $i = 1, 2, 3$. Write down the events in terms of $A_1, A_2$ and $A_3$ that:
  
\begin{parts}
    \item at least one of the tosses turns up with even number.
    \item exactly one of the tosses turns up with even number.
    \item not more than two of the tosses will turn up with even number.
  \end{parts}

  \item If $S = \{x : 0 < x < 12\}$, $M = \{x : 1 < x < 9\}$ and $N = \{x : 0 < x < 5\}$, then find:
  
\begin{parts}
    \item $M \cup N$
    \item $M \cap N$
    \item $M^c \cap N^c$
  \end{parts}

  \item A lot consists of 10 good articles, 4 with minor defects, and 2 with major defects. One article is chosen at random. Find the probability that:
  
\begin{parts}
    \item it has no defects,
    \item it has no major defects,
    \item it is either good or has major defects.
  \end{parts}

  \item The moment generating function of two independent random variables $X$ and $Y$ are:
  \[ M_X(t) = \frac{1}{27}(e^t + 2)^3 \quad  \text{and} \quad M_Y(t) = \left(\frac{1}{3}e^t + \frac{2}{3}\right)^5 \]
  respectively. If $Z = X + Y$, then Find:
  
\begin{parts}
    \item $P(Z < 2)$
    \item $P(0 \le Z \le 2 \mid Z \ge 1)$
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 2.}

\begin{parts}
  \item Define the terms sample space, elementary events, mutually exclusive events and exhaustive events. Write sample space in each of following cases and categorise it as discrete or continuous sample space:
  
\begin{parts}
    \item Toss a coin repeatedly until head appears first time.
    \item A light bulb is manufactured. It is then tested for its life length by inserting it into a socket and the time elapsed (in hours) until it burns out is recorded.
    \item Toss a coin four times and observe the sequence of heads and tails obtained.
  \end{parts}

  \item A man draws a ball from an urn containing 4 white and 2 red balls. If the ball is white, he does not return it to urn; if the ball is red, he does return it. He draws another ball. Let $A$ be the event that the first ball drawn is white, and let $B$ be the event that second ball drawn is white. Which of the following statement is true or false? Give reasons:
  
\begin{parts}
    \item $P(B) = \dfrac{3}{5}$
    \item $P(B \mid A) = \dfrac{3}{5}$
    \item $P(A \mid B) = \dfrac{9}{14}$
    \item $A$ and $B$ are independent
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 3.}

\begin{parts}
  \item A person visits randomly to a family of three children and counts the number of male children. If $X$ denotes the number of male children and it is known that female child is twice more probable than male then write probability distribution of $X$. Draw a graph of probability distribution and cumulative probability distribution of $X$.

  \item Let the pdf of $X$ be given by:
  \[ f(x) = 
\begin{cases} 1/3 &  \text{if } 0 < x < 1 \\ 2/3 &  \text{if } 2 < x < 3 \\ 0 &  \text{otherwise} \end{cases} \]
  Find the distribution function of $X$. Calculate:
  
\begin{parts}
    \item $P\left(|X| < \dfrac{3}{2}\right)$
    \item $P[(2x + 1) < 4]$
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 4.}

\begin{parts}
  \item Let joint pmf of two random variables $(X, Y)$ be given by:
  \[ P(x, y) = 
\begin{cases} 1/6 &  \text{if } (x,y) = (-1,-1), (-1,1), (1,-1), (1,1) \\ 1/12 &  \text{if } (x,y) = (0,-1), (-1,0), (0,1), (1,0) \\ 0 &  \text{otherwise} \end{cases} \]
  Find the correlation coefficient between $X$ and $Y$.

  \item Define expectation of a random variable. Show that $E(X + Y) = E(X) + E(Y)$. Suppose that $X$ is a continuous random variable with the following pdf:
  \[ f(x) = 
\begin{cases} \frac{e^x}{2} &  \text{if } x \le 0 \\ \frac{e^{-x}}{2} &  \text{if } x > 0 \\ 0 &  \text{otherwise} \end{cases} \]
  Find expectation of $Y = |X|$.
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 5.}

\begin{parts}
  \item Define moment generating function. Obtain moment generation function and hence mean of a random variable $X$ having pdf:
  \[ f(x) = 
\begin{cases}  \theta e^{- \theta x} &  \text{if } x \ge 0 \\ 0 &  \text{otherwise} \end{cases} \]

  \item Define probability generating function. Obtain probability generating function for following distributions:
  
\begin{parts}
    \item $p(x) = \dbinom{n}{x} p^x (1-p)^{n-x}, \quad x = 0, 1, 2, \dots, n; \quad 0 < p < 1$
    \item $p(x) = \dfrac{e^{-\lambda} \lambda^x}{x!}, \quad x = 0, 1, 2, \dots, \infty; \quad \lambda > 0$
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 6.}

\begin{parts}
  \item Write the statement of Chebyshev's inequality and provide its proof. Let $X$ be a random variable with $E(X) = 2$ and $\operatorname{Var}(X) = 4$. Find lower bound of $P(-4 \le X \le 8)$.

  \item Define normal distribution and show that its median and modes are same. In a group of individuals characteristics $X \sim N(5, 4)$. If we select 2 individuals at random then what is probability that the value $X$ is greater than 5 for exactly one of them.
\end{parts}

\medskip\rule{\linewidth}{0.2pt}\medskip




\noindent   \textbf{Question 7.}

\begin{parts}
  \item Let $X \sim b(n, p)$. Show that:
  \[ \mu_{r+1} = pq \left( n r \mu_{r-1} + \frac{d}{dp}\mu_r \right) \]
  If $P(X = 0) = P(X = 1)$ and variance of $X$ is $1/4$, then find the value of $n$ and $q = 1 - p$.

  \item Show that Poisson distribution is limiting case of binomial distribution. If $X$ and $Y$ are independent Poisson variate with $P(X=1) = P(X=2)$ and $P(Y=2) = P(Y=3)$, then find variance of $X - Y$.
\end{parts}
