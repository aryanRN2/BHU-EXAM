\begin{parts}
  \item Define moment generating function (MGF) and cumulative generating function (CGF).\pts{3}

  \item Define Normal distribution. Write down its PDF and sketch the normal curve.\pts{3}
  \item Define Chi-square distribution and state its chief properties.\pts{2.5}
  \item Define Student's $t$-distribution and $F$-distribution.\pts{2.5}
\end{parts}
