\noindent{\large\bfseries SECTION --- A}
\rule{\linewidth}{0.4pt}\smallskip

\noindent\textbf{Question 1.} Attempt all parts: \hfill \textbf{10 $\times$ 1 = 10 Marks}
\begin{parts}
  \item Which of the following is a characteristic of PaaS?
  \begin{enumerate}[label=(\alph*)]
    \item Supports only one programming language
    \item Does not integrate with web services
    \item Provides an ability to ``Auto-scale''
    \item Requires manual hardware updates
  \end{enumerate}
  
  \item PaaS is primarily used for:
  \begin{enumerate}[label=(\alph*)]
    \item Networking hardware
    \item Developing, testing, running, and managing applications
    \item Providing email services
    \item Hosting web pages
  \end{enumerate}
  
  \item Which model involves the infrastructure residing at the customer's premises?
  \begin{enumerate}[label=(\alph*)]
    \item Public cloud
    \item Private cloud
    \item Hybrid cloud
    \item Community cloud
  \end{enumerate}
  
  \item Which of these is an example of an IaaS provider?
  \begin{enumerate}[label=(\alph*)]
    \item Google App Engine
    \item Amazon Web Services (AWS)
    \item Salesforce
    \item Slack
  \end{enumerate}
  
  \item SaaS applications are:
  \begin{enumerate}[label=(\alph*)]
    \item Hosted locally on user devices
    \item Accessible over the internet
    \item Only for enterprise users
    \item Managed by end-users
  \end{enumerate}
  
  \item Which service model allows users to avoid the cost and complexity of purchasing and managing physical servers?
  \begin{enumerate}[label=(\alph*)]
    \item SaaS
    \item IaaS
    \item PaaS
    \item NaaS
  \end{enumerate}
  
  \item In which model do users have control over operating systems, storage, and deployed applications?
  \begin{enumerate}[label=(\alph*)]
    \item SaaS
    \item PaaS
    \item IaaS
    \item NaaS
  \end{enumerate}
  
  \item SaaS applications often integrate with:
  \begin{enumerate}[label=(\alph*)]
    \item On-premise software only
    \item Other cloud-based services
    \item Local hardware solutions
    \item None of the above
  \end{enumerate}
  
  \item Which deployment model offers resources over the internet to anyone willing to pay?
  \begin{enumerate}[label=(\alph*)]
    \item Public cloud
    \item Private cloud
    \item Hybrid cloud
    \item Community cloud
  \end{enumerate}
  
  \item PaaS is advantageous for integration because it:
  \begin{enumerate}[label=(\alph*)]
    \item Requires less manual coding
    \item Provides pre-configured development environments
    \item Eliminates the need for middleware
    \item Reduces application complexity
  \end{enumerate}
\end{parts}
