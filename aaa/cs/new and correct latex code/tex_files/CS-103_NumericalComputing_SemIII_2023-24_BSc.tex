\noindent   \textbf{Question 2.} For the following system of linear equations:
\[

\begin{aligned}
  9x_1 + 2x_2 + 4x_3 &= 20 \\
  x_1 + 10x_2 + 4x_3 &= 6 \\
  2x_1 - 4x_2 + 10x_3 &= -15
\end{aligned}
\]

\begin{parts}
  \item Check the convergence of the iterative procedure.\pts{4}
  \item Use the Gauss-Seidel iterative procedure to solve the above (correct to 3 significant digits).\pts{10}
\end{parts}
