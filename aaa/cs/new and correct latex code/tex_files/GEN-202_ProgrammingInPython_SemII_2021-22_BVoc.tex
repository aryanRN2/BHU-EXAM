\medskip\noindent\textbf{Question 1.} What will be the output of the following code snippet?
\begin{lstlisting}[language=Python]
a = 3
b = 1
print(a, b)
a, b = b, a
print(a, b)
\end{lstlisting}
\begin{enumerate}[label=(\alph*),itemsep=2pt]
  \item 3 1 1 3
  \item 3 1 3 1
  \item 1 3 1 1
  \item 1 3 3 1
\end{enumerate}

\medskip\noindent\textbf{Question 2.} What will be the result of the following command in Python?
\begin{lstlisting}[language=Python]
round(0.4) - round(-0.5)
\end{lstlisting}
\begin{enumerate}[label=(\alph*),itemsep=2pt]
  \item 0
  \item 1
  \item 2
  \item -1
\end{enumerate}

  \item None of the mentioned
