\noindent   \textbf{Question 1.} Answer any   \textbf{four} of the following:\pts{$4 imes4=16$}

\begin{parts}
  \item Find the graphical solution of the following LPP:
  
\begin{align*}
     \text{Maximize } Z &= 2x_1 + 2x_2 \\
     \text{Subject to:}\quad x_1 + x_2 &\leq 7 \\
    0.5x_1 + x_2 &\leq 2 \\
    x_1,\, x_2 &\geq 0
  \end{align*}
  Is it a well-defined LPP?

  \item How do you calculate the mean and variance of the earliest occurrence time $T_E$ of event $i$ in a project arrow diagram, given $a_i$, $b_i$, and $m_i$ as the optimistic, pessimistic, and most likely times respectively?

  \item What are slack, surplus, and artificial variables? Explain their use for finding the initial basic feasible solution of a LPP.

  \item What is a non-linear programming problem? Define local and global minima and maxima of a single-variable non-linear function $f(x)$.

  \item Define the following with an example of each:
  
\begin{enumerate}[label=(\roman*)]
    \item A directed and an undirected cycle
    \item A connected network
    \item An undirected path and an undirected network
  \end{enumerate}
\end{parts}

\begin{parts}
  \item Explain the steps of the Simplex method for solving a LPP.
  \item Solve the following LPP using the Simplex method:
  
\begin{align*}
     \text{Maximize } Z &= 3x_1 + 5x_2 + x_3 \\
     \text{Subject to:}\quad 2x_1 + 3x_2 &\leq 3 \\
    x_2 + x_3 &\leq 8 \\
    3x_1 + 2x_2 + 4x_3 &\leq 15 \\
    x_1,\, x_2,\, x_3 &\geq 0
  \end{align*}
\end{parts}
