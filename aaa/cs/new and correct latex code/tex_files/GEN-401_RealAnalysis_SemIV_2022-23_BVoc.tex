\noindent {\large\bfseries Section A --- Long Answer Questions}\pts{$12 imes2=24$}

\medskip


\noindent   \textit{Attempt any two questions. Each question carries 12 marks.}

\bigskip




\noindent   \textbf{Question 1.} Calculate the divergence and curl of the vector field $\mathbf{F} = x^2\,\hat{i} + xy\,\hat{j} + yz\,\hat{k}$. Interpret the physical meaning of these operators.

\bigskip




\noindent   \textbf{Question 2.} Prove that $f(x) = 2x + 3$ is continuous at $x = 1$.

\bigskip




\noindent   \textbf{Question 3.} Find the first and second order partial derivatives of the function $f(x,y) = 3x^2 + 4xy - 5y^2$.

\bigskip




\noindent   \textbf{Question 4.} Calculate the gradient of $f(x,y,z) = x^2 + xy + z^2$ and evaluate it at the point $(1,1,1)$.

\bigskip

\rule{\linewidth}{0.4pt}



\noindent {\large\bfseries Section B --- Short Answer Questions}\pts{$6 imes6=36$}

\medskip


\noindent   \textit{Answer any six out of the following nine questions. Each question carries 6 marks.}

\bigskip




\noindent   \textbf{Question 5.} Compute the curl of the vector field $\mathbf{G} = \sin(x)\,\hat{i} + \cos(y)\,\hat{j}$.

\bigskip




\noindent   \textbf{Question 6.} Prove that $f(x) = x^2$ is continuous but not uniformly continuous on $\mathbb{R}$.

\bigskip




\noindent   \textbf{Question 7.} Calculate the directional derivative of $f(x,y) = x^2 + y^2$ at the point $(1,1)$ in the direction of $\hat{i} + \hat{j}$.

\bigskip




\noindent   \textbf{Question 8.} Prove that the function $f(x,y) = x^2$ is differentiable at $x = 0$.

\bigskip




\noindent   \textbf{Question 9.} Use the limit definition of a derivative to find $f'(x)$ for $f(x) = x^2$.

\bigskip




\noindent   \textbf{Question 10.} Compute $\dfrac{\partial^2}{\partial x\,\partial y}(3x^2 + 4xy - 5y^2)$.

\bigskip




\noindent   \textbf{Question 11.} Calculate the unit vector perpendicular to both $\mathbf{A} = 2\hat{i} - \hat{j} + 2\hat{k}$ and $\mathbf{B} = \hat{i} + \hat{j} - \hat{k}$.

\bigskip




\noindent   \textbf{Question 12.} Compute the limit $\displaystyle\lim_{x  o 2} \frac{x^2 - 4}{x - 2}$.

\bigskip




\noindent   \textbf{Question 13.} Show that the function $f(x) = x^2$ is not uniformly continuous on $\mathbb{R}$.

\bigskip
