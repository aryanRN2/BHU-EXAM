\noindent   \textbf{Question 1.} Answer the following multiple choice questions:

\begin{parts}
  \item The correct sequence of HTML tags for starting a webpage is:
  
\begin{parts}
    \item   \texttt{<html>},   \texttt{<head>},   \texttt{<title>},   \texttt{<body>}
    \item   \texttt{<head>},   \texttt{<title>},   \texttt{<html>},   \texttt{<body>}
    \item   \texttt{<html>},   \texttt{<body>},   \texttt{<title>},   \texttt{<head>}
    \item   \texttt{<html>},   \texttt{<head>},   \texttt{<body>},   \texttt{<title>}
  \end{parts}

  \item What will happen if the following JavaScript code is executed?
  
\begin{lstlisting}
var count = 0;
while (count < 10) {
    console.log(count);
    count++;
}
  \end{lstlisting}
  
\begin{parts}
    \item An error is displayed.
    \item An exception is thrown.
    \item The values of count variable are stored in a particular location.
    \item The values of count from 0 to 9 are displayed in the console.
  \end{parts}

  \item How to create a hyperlink in HTML?
  
\begin{parts}
    \item   \texttt{<a href="http://www.javatpoint.com">javaTpoint.com</a>}
    \item   \texttt{<a url="www.javatpoint.com">javaTpoint.com</a>}
    \item   \texttt{<a link="www.javatpoint.com">javaTpoint.com</a>}
    \item   \texttt{<a>www.javatpoint.com</a>}
  \end{parts}

  \item How to create a checkbox in HTML?
  
\begin{parts}
    \item   \texttt{<input type="checkbox">}
    \item   \texttt{<input type="button">}
    \item   \texttt{<checkbox>}
    \item   \texttt{<input type="check">}
  \end{parts}

  \item Which of the following is the correct way to start an ordered list with a numeric value of 4?
  
\begin{parts}
    \item   \texttt{<ol type="1" start="4">}
    \item   \texttt{<ol type="1" initial="4">}
    \item   \texttt{<ol type="1" begin="4">}
    \item   \texttt{<ol type="1" num="4">}
  \end{parts}

  \item Which of the following is the correct syntax to make the background-color of all paragraph elements yellow?
  
\begin{parts}
    \item   \texttt{p \{background-color: yellow;\}}
    \item   \texttt{p \{background-color: \#yellow;\}}
    \item   \texttt{all \{background-color: yellow;\}}
    \item   \texttt{all p \{background-color: \#yellow;\}}
  \end{parts}

  \item The CSS property used to specify the transparency of an element is:
  
\begin{parts}
    \item   \texttt{opacity}
    \item   \texttt{filter}
    \item   \texttt{visibility}
    \item   \texttt{overlay}
  \end{parts}

  \item Which of the following syntax is correct in CSS to make each word of a sentence start with a capital letter?
  
\begin{parts}
    \item   \texttt{text-transform: capitalize;}
    \item   \texttt{text-style: capital;}
    \item   \texttt{transform: capitalize;}
    \item   \texttt{text-transform: capital;}
  \end{parts}

  \item In JavaScript, the   \texttt{x === y} statement implies that:
  
\begin{parts}
    \item Both x and y are equal in value and data type.
    \item Both x and y are equal in value, type, and reference address.
    \item Both x and y are equal in value only.
    \item Both are not the same.
  \end{parts}
\end{parts}
