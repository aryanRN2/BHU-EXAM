\medskip


\noindent   \textbf{Question 1.}

\begin{parts}
  \item Explain five major types of programming languages with examples. \pts{5}
  \item What is the difference between the top-down and bottom-up approaches of programming? \pts{5}
  \item What are the steps involved in solving a problem? Explain your answer with an example. \pts{4}
\end{parts}

\medskip


\noindent   \textbf{Question 2.}

\begin{parts}
  \item Write an algorithm to find the greatest among three numbers. \pts{4}
  \item Design an algorithm and flowchart to convert a decimal number to a binary number. \pts{6}
  \item Compare and contrast compilers and interpreters. \pts{4}
\end{parts}

\medskip


\noindent   \textbf{Question 3.}

\begin{parts}
  \item Describe the order of precedence with regards to operators in the C language. \pts{5}
  \item Explain the different data types used in the C language. \pts{5}
  \item Write the steps involved in converting C source code to executable code. \pts{4}
\end{parts}

\medskip


\noindent   \textbf{Question 4.}

\begin{parts}
  \item Write a C program to find the factorial of a number using recursion. \pts{6}
  \item What will be the outcome of the following conditional statement if the value of variable   \texttt{s} is 10? \\
   \texttt{s >= 10 \&\& s < 25 \&\& s != 12} \pts{4}
  \item Write a call-by-value function to swap two numbers. \pts{4}
\end{parts}

\medskip


\noindent   \textbf{Question 5.}

\begin{parts}
  \item Explain with examples pre-increment (   \texttt{++i}) and post-increment (   \texttt{i++}). \pts{4}
  \item What is the difference between formal arguments and actual arguments? \pts{3}
  \item What are variables, and in what way are they different from constants? \pts{4}
  \item Explain call-by-value and call-by-reference. \pts{3}
\end{parts}

\medskip


\noindent   \textbf{Question 6.}

\begin{parts}
  \item Write the differences between insertion sort and selection sort. \pts{4}
  \item Write the pseudocode of the quicksort algorithm. \pts{5}
  \item What will be the output of the following C code?
  
\begin{lstlisting}[language=C]
#include <stdio.h>
int main() {
    int a = 1, b = 2, c = 3, d = 4, e;
    e = c + d - b * a;
    printf("%d, %d
", e, d);
    return 0;
}
  \end{lstlisting} \pts{5}
\end{parts}

\medskip


\noindent   \textbf{Question 7.}

\begin{parts}
  \item How does insertion sort work? Write the pseudocode of insertion sort. Explain step-by-step how to sort an array $A = [3, 8, 5, 4, 1, 9, -2]$ in descending order. \pts{7}
  \item What is a searching algorithm? Explain the different types of search operations. \pts{3}
  \item Explain step-by-step how to search for the element '4' in array $A = [3, 8, 5, 4, 1, 9, -2]$ using the binary search algorithm. \pts{4}
\end{parts}
