\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Show that solutions of $n^{th}$-order differential equations may contain $n$ arbitrary constants.
  \item Solve the BVP: $\dfrac{d^2y}{dx^2}+y=0$, $y(0)=1$, $y(\pi/2)=0$.
  \item Separate the variables in $\dfrac{dy}{dx}=e^{x-y}+x^2 e^{-y}$ and solve.
  \item Solve by Bernoulli's equation: $\dfrac{dy}{dx}-\dfrac{y}{x}=y^2\sin x$.
  \item Define an indicial equation in ODE.
  \item Show that $P_1(x)=x$ where $P_n(x)$ is the Legendre polynomial.
  \item Show that $\dfrac{d}{dx}[x^n J_n(x)]=x^n J_{n-1}(x)$.
\end{parts}
