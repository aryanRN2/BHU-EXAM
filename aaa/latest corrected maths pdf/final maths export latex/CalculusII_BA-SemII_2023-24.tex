\noindent\textbf{Question 1.} \textit{(Compulsory --- 2 marks each.)}
\begin{parts}
  \item Determine the repeated limits of $f(x, y) = \frac{x-y}{x+y}$ at $(0, 0)$.
  \item By using the definition of limit, show that $\lim_{(x,y) \to (1,2)} xy = 2$.
  \item If $z = x^2 y^3$, where $x = t^2$ and $y = t$, then find $\frac{dz}{dt}$.
  \item Investigate for maximum and minimum values the function given by $f(x) = \sin x + \cos 2x$.
  \item If $f(x,y) = \begin{cases} \frac{xy}{\sqrt{x^2+y^2}}, & \text{if } x^2+y^2 \neq 0 \\ 0, & \text{if } x=y=0 \end{cases}$, then find $f_x(0,0)$ and $f_y(0,0)$.
  \item Verify Euler's theorem for the function $u = x^3 \log(y/x)$.
  \item Show that $\beta(m, n) = \frac{\Gamma(m)\Gamma(n)}{\Gamma(m+n)}$.
  \item If $u^u v^v w^w = 1$, then show that $\frac{\partial^2 u}{\partial v \partial w} = -(u \log (eu))^{-1}$ when $u = v = w$.
  \item Show that $\int_0^\infty e^{-x^2} \, dx = \frac{\sqrt{\pi}}{2}$.
  \item Evaluate: $\int_1^e \int_0^{\log y} \int_1^{e^x} \log z \, dz \, dx \, dy$.
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item Show that the function $f(x, y) = \sqrt{|xy|}$, where $(x, y) \in \mathbb{R}^2$, is continuous but not differentiable at the origin. \pts{6}
  \item State and prove a sufficient condition for differentiability of a function of two variables. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item State and prove Schwarz's theorem for a function of two variables. \pts{6}
  \item Expand the function $f(x, y) = x^2y + 3y - 2$ by Taylor's theorem in powers of $(x - 1)$ and $(y + 2)$. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 4.}
\begin{parts}
  \item If $u = \tan^{-1}\left( \frac{xy}{\sqrt{1+x^2+y^2}} \right)$, then show that $\frac{\partial^2 u}{\partial x \partial y} = \frac{1}{(1+x^2+y^2)^{3/2}}$. \pts{6}
  \item State and prove Euler's theorem for homogeneous functions. If $u = \tan^{-1}\left( \frac{x^3+y^3}{x-y} \right)$, $x \neq y$, then show that:
  \[
  x \frac{\partial u}{\partial x} + y \frac{\partial u}{\partial y} = \sin 2u
  \] \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 5.}
\begin{parts}
  \item Determine the points where the function $f(x, y) = x^4 + y^4 - 2x^2 + 4xy - 2y^2$ has a maximum or minimum. \pts{6}
  \item Prove that of all rectangular parallelopipeds of a given volume, the cube has the least surface area. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 6.}
\begin{parts}
  \item Change the order of integration in the double integral:
  \[
  \int_0^{2a} \int_{\sqrt{2ax-x^2}}^{\sqrt{2ax}} f(x,y) \, dy \, dx \quad (a > 0)
  \] \pts{6}
  \item Transform the integral $\int_0^1 \int_0^{\sqrt{1-x^2}} y^2 (x^2+y^2) \, dy \, dx$ by changing to polar coordinates, and hence evaluate it. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\begin{parts}

  \iiint x^{l-1} y^{m-1} z^{n-1} \, dx \, dy \, dz
  \]

\end{parts}
