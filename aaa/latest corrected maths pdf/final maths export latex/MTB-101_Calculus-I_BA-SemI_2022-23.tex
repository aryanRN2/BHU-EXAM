\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Define a bounded sequence with a suitable example.
  \item Explain convergent and divergent sequences with suitable examples.
  \item Prove that if $\displaystyle\sum_{n=1}^{\infty}u_n$ converges, then $\lim_{n o\infty}u_n=0$.
  \item State D'Alembert's ratio test for $\displaystyle\sum_{n=1}^{\infty}u_n$ of positive real numbers.
  \item Find the asymptotes of $4x^2+9y^2=x^2y^2$.
  \item Find the point(s) of inflection of $f(x)=x^4+2x^3$.
  \item Find the $n$th order derivative of $y=\sin(ax+b)$, $a
eq0$, $b
eq0$.
  \item Determine the critical points of $f(x)=6+12x+3x^2-2x^3-4x^4$, $x\in\mathbb{R}$.
  \item Elaborate the Sandwich theorem for sequences.
  \item Show that $\displaystyle\sum_{n=1}^{\infty}\dfrac{(-1)^{n+1}}{n}$ is convergent.
  \item State Taylor's theorem with Lagrange's form of remainder.
\end{parts}
