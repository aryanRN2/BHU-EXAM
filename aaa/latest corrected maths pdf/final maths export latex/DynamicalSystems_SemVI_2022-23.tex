\noindent\textbf{Question 1.} \textit{(Compulsory --- 2 marks each.)}
\begin{parts}
  \item Find the evolution operator $\phi_t$ for $\dot{x} = x^2 - 1$.
  \item Show that all solutions except the fixed point at origin of the system $\dot{x} = y$, $\dot{y} = -x$ are periodic.
  \item If $P$, $Q$ and $R$ are matrices of order $n$ such that $R = P Q P^{-1}$, then show that $e^R = P e^Q P^{-1}$.
  \item For $A = \begin{pmatrix} \lambda & 1 \\ 0 & \lambda \end{pmatrix}$, find the exponential matrix $e^{At}$.
  \item Is the matrix $\Phi(t) = \begin{pmatrix} 2e^t & -e^{-t} \\ e^t & e^{-t} \end{pmatrix}$ a fundamental matrix for the system $\dot{x} = Ax$? If so, find the matrix $A$.
  \item Calculate $e^{At}$ for $A = \begin{pmatrix} 0 & 4 \\ 4 & 0 \end{pmatrix}$.
  \item Does the non-autonomous equation $\ddot{x} + \left(\frac{2}{t}\right)\dot{x} + x = 0$ ($t > 0$) have bounded solutions?
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item State the Lipschitz condition for an initial value problem $\dot{x} = f(t, x)$, $x(t_0) = x_0$. Find the maximum interval of existence for the unique solution of the system $\dot{x} = x^2 + \cos^2 t$ for $t > 0$, $x(0) = 0$. \pts{6}
  \item Find the evolution operator $\phi_t$ for the one-dimensional flow $\dot{x} = -x^3$ and show that it forms a dynamical group. \pts{8}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item Solve the system $\dot{x}_1 = x_1 - 2x_2$, $\dot{x}_2 = -3x_1 + 2x_2$ by calculating the fundamental matrix of the system. \pts{7}
  \item Find the analytical solution of the system:
  \[
  \dot{x} = -y + x(1-x^2-y^2), \quad \dot{y} = x + y(1-x^2-y^2)
  \]
  by changing it to equivalent polar form. \pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 4.}
\begin{parts}
  \item Define an invariant set $S$ for a dynamical system $\dot{x} = f(x)$. Show that the line $y = mx$ is invariant for the system $\dot{x} = Ax$, where $A$ is a constant $2\times 2$ matrix with a simple real eigenvalue $\lambda$ and corresponding eigenvector $(1, m)^T$, $m \in \mathbb{R}$. \pts{7}
  \item Determine the stable and unstable subspaces for the linear system $\dot{x} = Ax$, where
  \[
  A = \begin{pmatrix} -2 & -1 & 0 \\ 1 & -2 & 0 \\ 0 & 0 & 3 \end{pmatrix}
  \]
  and show that these subspaces are invariant. \pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 5.}
\begin{parts}
  \item If
  \[
  A = \begin{pmatrix} a & -b \\ b & a \end{pmatrix}
  \]
  then show that $e^{At} = e^{at} \left[ I \cos(bt) + J \sin(bt) \right]$, where $I = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}$ and $J = \begin{pmatrix} 0 & -1 \\ 1 & 0 \end{pmatrix}$. \pts{7}
  \item Solve the initial value problem $\dot{x} = Ax$, $x(0) = x_0$, with
  \[
  A = \begin{pmatrix} 0 & -2 & -1 & -1 \\ 1 & 2 & 1 & 1 \\ 0 & 1 & 1 & 0 \\ 0 & 0 & 0 & 1 \end{pmatrix}
  \]
  by using the $A = S + N$ method, where $N$ is a nilpotent matrix. \pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 6.}
\begin{parts}
  \item Determine all fixed points of the nonlinear system $\dot{x} = y$, $\dot{y} = x - x^3 - y$ and discuss their stability by linearizing the system. \pts{7}
  \item Consider the non-autonomous differential equation:
  \[
  \dot{x} = -x + \left( \frac{a}{1+t} \right) x^2, \quad a > 0, \ t \ge 0
  \]
  Discuss the stability of the critical point $x = 0$ for various values of $a$. \pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 7.}
\begin{parts}
  \item Discuss the existence and stability of all fixed points of the system:
  \[
  \dot{x} = x(1-x-ay), \quad \dot{y} = b(x-c)y
  \]
  where $a$, $b$, and $c$ are non-negative constants. \pts{7}
  \item Find the solution of the non-homogeneous system:
  \[
  \dot{x} = x + y + t, \quad \dot{y} = -y + 1
  \]
  with the initial conditions $x(0) = 1$, $y(0) = 0$. \pts{7}
\end{parts}
