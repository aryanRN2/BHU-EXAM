\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Let $f(x,y)=\dfrac{xy}{\sqrt{x^2+y^2}}$ for $(x,y)
eq(0,0)$. Check whether $\displaystyle\lim_{(x,y) o(0,0)}f(x,y)$ exists.
  \item Let $f(x,y)=
\begin{cases}\dfrac{x^2y^2}{x^2+y^2} & (x,y)
eq(0,0) \\ 0 & x=y=0\end{cases}$. Find $f_x$ and $f_y$ at the origin and test continuity of $f_x$ at the origin.
  \item If $u=\sin(x^2+y^2)$ where $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$, find $\dfrac{du}{dx}$.
  \item Define a homogeneous function of degree $n$ in four variables $x,y,z,w$ and give an example.
  \item Let $U$ be a non-empty open set in $\mathbb{R}^2$ and $f:U o\mathbb{R}$. Give the definition of the derivative of $f$ at $(a,b)\in U$. Find the derivative of $f(x,y)=x^2y+xy^2$ at $(-1,-1)$.
  \item Change the order of integration in $\displaystyle\int_0^{\infty}dy\int_y^{\infty}e^{-x^2}dx$ and evaluate it.
  \item Find the volume of the solid bounded by $x^2+y^2=1$ and the planes $y+z=1$ and $z=0$.
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 2.}

\begin{parts}
  \item Let $f,g:D o\mathbb{R}$ with $\lim_{(x,y) o(a,b)}f(x,y)=\ell$ and $\lim_{(x,y) o(a,b)}g(x,y)=m$. Show that $\lim_{(x,y) o(a,b)}f(x,y)g(x,y)=\ell m$.\pts{5}
  \item State and prove Young's theorem for a function of two variables. Verify with an example.\pts{9}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 3.}

\begin{parts}
  \item Let $f:D o\mathbb{R}$ be differentiable at $(a,b)\in D$. Show that $f$ is continuous at $(a,b)$. Give an example of $f(x,y)$ continuous at $(0,0)$ but not differentiable there.\pts{6}
  \item Let $C\subseteq\mathbb{R}^2$ with $(a,b)\in \text{int}(C)$ and $f:C o\mathbb{R}$. Suppose $\exists\, r>0$ such that (i) $B_r((a,b))\subseteq C$; (ii) $f_x(x,y)$ exists $\forall\,(x,y)\in B_r((a,b))$ and $f_x$ is continuous at $(a,b)$; (iii) $f_y$ exists at $(a,b)$. Show that $f$ is differentiable at $(a,b)$.\pts{8}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 4.}

\begin{parts}
  \item State and prove Euler's theorem for homogeneous functions of degree $n$ in $x,y,z$. Verify for a homogeneous function of degree 3.\pts{5}
  \item Prove that $\dfrac{\partial^2 f}{\partial x^2}+\dfrac{\partial^2 f}{\partial y^2}$ is invariant under change of rectangular axes.\pts{5}
  \item Show that $u=x+3y+2z$, $v=3x+4y-2z$, $w=11x+18y-2z$ are not independent and find the relation among them.\pts{4}
\end{parts}
