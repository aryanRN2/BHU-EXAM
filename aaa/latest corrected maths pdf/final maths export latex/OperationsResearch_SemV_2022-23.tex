\noindent   \textbf{Question 1.}

\begin{parts}
  \item Solve graphically: Maximize $3x + 4y$ subject to $x + y \le 1$, $x \ge y$, $x, y \ge 0$. \pts{2}
  \item Find a basic feasible solution of: Minimize $[1,\,-2]
\begin{bmatrix}x_1\\x_2\end{bmatrix}$ subject to $
\begin{bmatrix}1 & -1\\-1 & 1\end{bmatrix}
\begin{bmatrix}x_1\\x_2\end{bmatrix} \ge 
\begin{bmatrix}2\\-1\end{bmatrix}$. \pts{2}
  \item Find an extreme point of $X = \{(x,y): x + y \le 1,\, 0 \le x,y \le 1\}$. \pts{2}
  \item Consider a standard form LPP with $A = 
\begin{bmatrix}0 & 2 & 0 & 1\\1 & 1 & 0 & 0\\0 & 0 & 3 & 0\end{bmatrix}$, $b = 
\begin{bmatrix}7\\8\\9\end{bmatrix}$, $c = 
\begin{bmatrix}4\\0\\5\end{bmatrix}$. Suppose the reduced cost coefficient vector for some basis is $ar{c}^T = [0, 1, 0, 0]$. Find an optimal solution to minimize the problem. \pts{2}
  \item Using part (d), find the value of the second cost coefficient $a$. \pts{2}
  \item Write the dual of: Minimize $x$ subject to $x \le 1$. \pts{2}
  \item Prove that the set of all optimal solutions of an LPP is a convex set. Illustrate with an example. \pts{2}
\end{parts}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item State and prove the Fundamental Theorem of Linear Programming. \pts{7}
  \item Consider $Ax = b$ where $A = 
\begin{bmatrix}2 & 3 & -1 & -1\\1 & 0 & 2 & 1\end{bmatrix}$, $b = 
\begin{bmatrix}1\\1\end{bmatrix}$. Find a (non-basic) feasible solution of $Ax = b$ and use the method in part (a) to find a basic feasible solution. \pts{7}
\end{parts}

\noindent   \textbf{Question 3.}

\begin{parts}
  \item Let $X$ be the convex set of all feasible solutions of $Ax = b$, $x \ge 0$ where $A \in \mathbb{R}^{m  \times n}$, $m < n$. Show that $x$ is an extreme point of $X$ if and only if $Ax = b$, $x \ge 0$ with the non-zero components of $x$ forming linearly independent columns of $A$. \pts{7}
  \item Using the simplex method, solve: Maximize $7x + 6y$ subject to $2x + y \le 3$, $x + 4y \le 4$, $x, y \ge 0$. \pts{7}
\end{parts}

  \item Show that the original LPP has a basic feasible solution if and only if the associated artificial problem has an optimal feasible solution with objective function value zero. \pts{6}
