\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Derive the PDE by eliminating arbitrary constants from the equation of a sphere.
  \item Form a PDE by eliminating $\phi$ from $z=(x-y)\phi(x^2+y^2)$.
  \item Solve: $p an x+q an y= an z$.
  \item Find the complete integral of $pq=p+q$.
  \item Find the particular integral of $r+s-2t=e^{x+y}$.
  \item Solve: $(D-D')z=0$.

\end{parts}
