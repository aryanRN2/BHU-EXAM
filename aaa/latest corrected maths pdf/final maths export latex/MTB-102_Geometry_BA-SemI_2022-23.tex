\noindent\textbf{Question 1.} \textit{(Compulsory --- 2 marks each.)}
\begin{parts}
  \item Define (i)~focus, (ii)~directrix, (iii)~eccentricity, (iv)~latus rectum of a conic section.
  \item Show that the general equation of a straight line in polar coordinates is $p=r\cos(\alpha-\theta)$, where $(p,\alpha)$ is the foot of the perpendicular from the origin.
  \item Transform $ax+by+cz+d=0$ into normal form $\ell x+my+nz=p$.
  \item Define the angle between two planes.
  \item Find the image of $(1,3,4)$ in the plane $2x-y+z+3=0$.
  \item Define shortest distance between two lines.
  \item Show that a plane section of a sphere is a circle.
  \item Find the sphere through $x^2+y^2=a^2$, $z=0$ and the point $(a,b,\gamma)$.
  \item Define reciprocal cones.
  \item Define a right circular cylinder.
  \item Obtain the tangent plane to $ax^2+by^2=2cz$ at $(\alpha,\beta,\gamma)$.
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item Show that the polar equation of a parabola is $\ell/r=1+\cos\theta$.\pts{6}
  \item If $PSQ$ and $PS'R$ are two chords of an ellipse through foci $S$ and $S'$, prove $\tfrac{1}{SP}+\tfrac{1}{SQ}\cdot\tfrac{1}{S'P}+\tfrac{1}{S'R}$ is independent of the position of $P$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item Derive the equation of a plane in (i)~normal form $(\cos\alpha)x+(\cos\beta)y+(\cos\gamma)z=p$ and (ii)~intercept form.\pts{6}
  \item Find the plane through $(2,1,-1)$ orthogonal to $x-y+z=1$ and $3x+4y-2z=0$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 4.}
\begin{parts}
  \item Derive the condition for two lines to be coplanar.\pts{6}
  \item Show the plane containing $\tfrac{x}{b}+\tfrac{z}{d}=1$, $y=0$ parallel to $\tfrac{y}{c}+\tfrac{z}{d}=1$, $x=0$ is $\tfrac{x}{b}-\tfrac{y}{c}-\tfrac{z}{d}+1=0$ and find the shortest distance between them.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 5.}
\begin{parts}
  \item A plane through $(a,b,c)$ cuts the axes in $A,B,C$. Show the locus of the centre of the sphere through $O,A,B,C$ is $\tfrac{a}{x}+\tfrac{b}{y}+\tfrac{c}{z}=2$.\pts{6}
  \item If two spheres of radii $r_1$ and $r_2$ cut orthogonally, prove the radius of the common circle is $\dfrac{r_1 r_2}{\sqrt{r_1^2+r_2^2}}$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 6.}
\begin{parts}
  \item Find the cone with vertex at origin and base $x=a$, $y^2+z^2=b^2$. Show the section by a plane parallel to $XOY$ is a hyperbola.\pts{6}
  \item Find the cylinder with generators parallel to $z$-axis and guiding curve $x^2+y^2=z$, $x+y+z=1$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 7.}
\begin{parts}
  \item Show the locus of intersection of three mutually perpendicular tangent planes to $ax^2+by^2+cz^2=1$ is $a^{-1}+b^{-1}+c^{-1}=x^2+y^2+z^2$.\pts{6}
  \item Prove that through any given point five normals can be drawn to a paraboloid.\pts{6}
\end{parts}
