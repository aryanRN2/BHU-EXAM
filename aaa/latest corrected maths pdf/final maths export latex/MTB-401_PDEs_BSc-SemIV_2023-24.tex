\noindent\textbf{Question 1.} \textit{(Compulsory.)}
\begin{parts}
  \item Form the PDE by eliminating $h$ and $k$ from $(x-h)^2+(y-k)^2+z^2=r^2$.\pts{2}
  \item Form the PDE by eliminating arbitrary functions $f$ and $g$ from $z=f(x+at)+g(x-at)$.\pts{2}
  \item Define the order and degree of a PDE. Give an example.\pts{2}
  \item Find the complete integral of $z=px+qy+p^2+q^2$, where $p=\partial z/\partial x$, $q=\partial z/\partial y$.\pts{2}
  \item Solve $(D_x^2-4D_xD_y+4D_y^2)z=0$, where $D_x=\partial/\partial x$, $D_y=\partial/\partial y$.\pts{3}
  \item Find the general solution of $s+p-q=z$.\pts{3}
  \item Solve $x^2r-y^2t+xp-yq=0$ by reducing to linear PDE with constant coefficients.\pts{3}
  \item Solve $xr+p=9x^2y^3$.\pts{3}
  \item Check whether $3r+6s+4t=0$ is elliptic.\pts{2}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item Obtain the general integral of $pt+3q=5z+\tan(y-3x)$.\pts{5}
  \item Apply Charpit's method to find the complete integral of $px+qy+z=xq^2$.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item Find the general integral of $r+s-6t=y\cos x$.\pts{5}
  \item Reduce $x^2\dfrac{\partial^2 z}{\partial x^2}+2xy\dfrac{\partial^2 z}{\partial x\,\partial y}+y^2\dfrac{\partial^2 z}{\partial y^2}+x\dfrac{\partial z}{\partial y}=x^2y^2$ to linear PDE with constant coefficients and solve it.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 4.}
\begin{parts}
  \item Solve $(2D_x^2-5D_xD_y+2D_y^2)z=24(y-x)$.\pts{5}
  \item Solve $(D_x^2-D_xD_y-2D_y^2+2D_x+2D_y)z=e^{x+y}$.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 5.}
\begin{parts}
  \item Solve $ys-p=xy^2\cos(xy)$.\pts{5}
  \item Reduce to canonical form and solve: $\dfrac{\partial^2 z}{\partial x^2}+2\dfrac{\partial^2 z}{\partial x\,\partial y}+\dfrac{\partial^2 z}{\partial y^2}=0$.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 6.}
\begin{parts}
  \item Solve by Monge's method: $yr-2ys+t=p+q^2$.\pts{5}
  \item Solve $r+4s+t+rt-s^2=2$.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 7.}
\begin{parts}
  \item Use separation of variables to solve the heat equation $\dfrac{\partial u}{\partial t}=\dfrac{\partial^2 u}{\partial x^2}$, $0<x<1$, $t>0$, subject to $u(x,0)=x^2-x$, $u(0,t)=u(1,t)=0$.\pts{6}
  \item Solve the wave equation $\dfrac{\partial^2 u}{\partial t^2}=c^2\dfrac{\partial^2 u}{\partial x^2}$ using separation of variables.\pts{6}
\end{parts}
