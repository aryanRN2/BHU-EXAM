\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each unless specified otherwise.)}

\begin{parts}
  \item If $f(z)$ and $\overline{f(z)}$ are both analytic functions defined in a domain $D$, show that $f(z)$ is constant. \pts{2}
  \item Find the relation among the real constants $a, b, c$ so that the function $f(z) = (x + ay) + i(bx + cy)$ is entire. \pts{3}
  \item Find the image of the line $y = 2x$ under the mapping $w = (1+i)z + 2$. \pts{2}
  \item Find the bilinear transformation which maps the points $z = i, 2, -2$ into $w = i, 1, -1$ respectively. \pts{3}
  \item Let $(X,d)$ be a metric space and let $C$ be a nonempty subset of $X$. Let $d_C : X \rightarrow \mathbb{R}$ be defined by:
  \[ d_C(x) = \inf_{y \in C} d(x, y), \quad  \text{for all } x \in X \]
  Show that:
  \[ |d_C(x) - d_C(y)| \le d(x, y), \quad  \text{for all } x, y \in X \] \pts{3}
  \item Let $X = \{ \frac{1}{n} : n \in \mathbb{N} \} \cup \{0\}$ be equipped with the usual metric. Let $A = \{ \frac{1}{2n} : n \in \mathbb{N} \}$ and $B = \{ \frac{1}{2n+1} : n \in \mathbb{N} \}$. Find the distance $d(A, B)$. \pts{2}
  \item Let $Y$ be a subset of a metric space $(X,d)$ and let $C$ be a subset of $Y$. Show that $x \in Y$ is a limit point of $C$ in $Y$ if and only if $x$ is a limit point of $C$ in $X$. \pts{4}
  \item Let $\mathbb{Q}$ be the set of all rational numbers. Show that $\mathbb{Q}$ is not complete under the usual metric. \pts{3}
\end{parts}

  where $a$ is a complex number such that $|a| < 1$. \pts{6}
\end{parts}

\noindent   \textbf{Question 3.}

\begin{parts}
  \item Let $d$ and $d'$ be metrics on a nonempty set $X$. Show that:
  
\begin{parts}
    \item $d + d'$ is a metric on $X$;
    \item $\max\{d, d'\}$ is a metric on $X$.
  \end{parts} \pts{6}
  \item Let $G$ be a subset of a metric space $(X,d)$. Show that the following statements are equivalent: \pts{6}
  
\begin{parts}
    \item $G$ is open.
    \item $G$ is the union of open balls in $X$.
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 4.}

\begin{parts}
  \item Let $Y$ be a subspace of a metric space $(X, d)$ and $U$ a subset of $Y$. Show that $U$ is open in $(Y, d_Y)$ if and only if there exists an open set $G$ in $X$ such that $U = Y \cap G$. \pts{6}
  \item Let $C$ be a subset of a metric space $(X, d)$. If the closure $\overline{C} = C \cup C'$ (where $C'$ is the set of all limit points of $C$), show that: \pts{6}
  
\begin{parts}
    \item $\overline{C}$ is closed;
    \item $\overline{C}$ is the smallest closed subset of $X$ containing $C$.
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 5.}

\begin{parts}
  \item Let $C$ be a nonempty subset of a metric space $(X, d)$. Show that:
  \[ \operatorname{diam}(C) = \operatorname{diam}(\overline{C}) \] \pts{6}
  \item Show that the following metrics defined on $\mathbb{R}^n$ (where $x = (x_1, \dots, x_n)$ and $y = (y_1, \dots, y_n)$) are equivalent: \pts{6}
  
\begin{parts}
    \item $d_\infty(x, y) = \max_{1 \le i \le n} |x_i - y_i|$
    \item $d_1(x, y) = \sum_{i=1}^n |x_i - y_i|$
    \item $d_2(x, y) = \sqrt{\sum_{i=1}^n (x_i - y_i)^2}$
  \end{parts}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 6.}

\begin{parts}
  \item Let $X = \ell_\infty$ be the set of all bounded sequences of real numbers, and let $d$ be defined by:
  \[ d(x, y) = \sup_{n \in \mathbb{N}} |x_n - y_n|, \quad  \text{for } x, y \in X \]
  Show that $(X, d)$ is a complete metric space. \pts{6}
  \item State and prove Cantor's Intersection Theorem. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 7.}

\begin{parts}
  \item Let $(X,d)$ and $(Y, \rho)$ be metric spaces, and $f : X \rightarrow Y$ be a function. Show that the following statements are equivalent: \pts{6}
  
\begin{parts}
    \item $f$ is continuous at $x_0 \in X$.
    \item $\{f(x_n)\}_{n=1}^\infty$ converges to $f(x_0)$ in $Y$ whenever $\{x_n\}_{n=1}^\infty$ is a sequence in $X$ converging to $x_0$.
  \end{parts}
  \item Let $(X,d)$ and $(Y, \rho)$ be metric spaces, and $f : X \rightarrow Y$ a uniformly continuous function. Show that the image of a Cauchy sequence in $X$ under $f$ is again a Cauchy sequence in $Y$. \pts{6}
\end{parts}
