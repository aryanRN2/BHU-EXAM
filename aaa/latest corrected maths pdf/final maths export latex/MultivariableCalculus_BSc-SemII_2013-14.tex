\noindent\textbf{Question 1.}
\begin{parts}
  \item Give the $\varepsilon$-$\delta$ definition of the limit of a function $f:\mathbb{R}^2\to\mathbb{R}$ as $(x,y)\to(a,b)$. Using the definition, show that:
  \[
  \lim_{(x,y)\to(0,0)} \frac{x^2}{x^2+y^2}
  \]
  does not exist. \pts{3}
  \item Show that the simultaneous limit of $f(x,y) = \dfrac{xy}{x^2+y^2}$ does not exist at $(0,0)$, but that the iterated limits exist and are equal. \pts{2}
  \item Let $f(x,y)$ be a homogeneous function of degree $5$. Show that:
  \[
  x^2\frac{\partial^2 f}{\partial x^2} + 2xy\frac{\partial^2 f}{\partial x\,\partial y} + y^2\frac{\partial^2 f}{\partial y^2} = 20f
  \] \pts{2}
  \item State Taylor's theorem for functions of two variables. \pts{2}
  \item Evaluate $\int_0^\infty x^n e^{-x}\,dx$. \pts{2}
  \item Show that $\int_0^\infty e^{-x^2}\,dy = \Gamma\!\left(\dfrac{n+1}{2}\right)$, $n > 0$. \pts{2}
  \item Write a necessary condition for $f(x,y)$ to have an extremal value. \pts{2}
  \item If $x, y, z$ are positive and $h_1 \le x + y + z \le h_2$, then find:
  \[
  \iiint (x+y+z)\,x^m\,y^n\,z^l\,dx\,dy\,dz
  \] \pts{2}
  \item If $\vec{r}$ is a vector function of scalar $t$ and $\vec{a}$ is a constant vector, differentiate the following with respect to $t$:
  \begin{parts}
    \item $\vec{r} \cdot \vec{a}$
    \item $\vec{r} \times \vec{a}$
    \item $|\vec{r}|\,\dfrac{d\vec{r}}{dt}$
  \end{parts} \pts{3}
  \item Prove that $\operatorname{div}\,\vec{r} = 3$. \pts{2}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item Prove that the limit of the product of two functions of two variables equals the product of their limits. \pts{7}
  \item Show that the function
  \[
  f(x,y) = \begin{cases} \dfrac{xy}{\sqrt{x^2+y^2}}, & (x,y)\ne(0,0) \\ 0, & (x,y)=(0,0) \end{cases}
  \]
  is continuous at the origin. \pts{5}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item If $(a,b)$ is a point of the domain $D\subset\mathbb{R}^2$ of a real-valued function $f(x,y)$, such that (i) $f_x(a,b)$ exists, (ii) $f_y(x,y)$ is continuous at $(a,b)$, then prove that $f(x,y)$ is differentiable at $(a,b)$. \pts{7}
  \item Show that $f(x,y) = \sin x + \cos y$ is differentiable everywhere. \pts{5}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\begin{parts}

  \item Transform the integral $\displaystyle\int_0^a\!\int_0^{\sqrt{a^2-x^2}} \!\!\sqrt{x^2+y^2}\,dx\,dy$ by changing to polar coordinates and hence evaluate it. \pts{5}
\end{parts}
