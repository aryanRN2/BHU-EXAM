\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Find the least $m\in\mathbb{N}$ for the sequence $x_n=\dfrac{3n+1}{7n-4}$ such that $\left|\dfrac{3n+1}{7n-4}-\dfrac{3}{7}\right|<\varepsilon$ for all $n>m$.
  \item Check whether $\{x_n\}$ defined by $x_1=\sqrt{7}$, $x_{n+1}=\sqrt{7+x_n}$ is monotonically increasing or decreasing.
  \item Examine convergence of $\sum_{n=1}^{\infty}\dfrac{n}{n(n+1)(n+3)}$.
  \item If $\sum u_n$ converges with positive terms, prove $\sum u_{2n}$ converges.
  \item Let $f:I o\mathbb{R}$, $\lim_{x o c}f(x)=\ell$. Show $\lim_{x o c}|f(x)|=|\ell|$.
  \item Give an example of $f,g$ not continuous at $c$ but $fg$ continuous at $c$.
  \item $f:\mathbb{R} o\mathbb{R}$ satisfies $f(x+y)=f(x)f(y)$. If $f$ is differentiable at $0$, prove $f'(c)=f(c)f'(0)$.
  \item Let $f:I o\mathbb{R}$ be differentiable at $c$ with $f'(c)<0$. Show $f$ is decreasing at $c$.
  \item If $c_0+\dfrac{c_1}{2}+\dfrac{c_2}{3}+\cdots+\dfrac{c_n}{n+1}=0$, show $c_0+c_1x+\cdots+c_nx^n=0$ has a root in $(0,1)$.
  \item Let $y=e^x(\cos x+\sin x)$. Prove $y''+4y=0$.
  \item Draw the graph of $r=2\sin(2 \theta)$.
\end{parts}

\noindent   \textbf{Question 2.}

\end{parts}
