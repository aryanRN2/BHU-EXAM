\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Define Laplace transform. Evaluate $\mathcal{L}\{e^t\}$ for all $t>0$.
  \item Define a piecewise continuous function. Find the Laplace transform of a piecewise continuous function.
  \item What is a function of exponential order? Give an example of a function that is NOT of exponential order and prove it.
  \item Evaluate $\mathcal{L}^{-1}\!\left\{\dfrac{1}{s+1}\right\}$.
  \item Find $\mathcal{L}\{t(3\cos 2t-2\sin 2t)\}$.
  \item Find the Laplace transform of $\displaystyle\int_0^t\sin 2u\,du$.
  \item Explain the change-of-scale property of the Laplace transform.
  \item Differentiate between the initial and final value theorems.
  \item Find the Laplace transform of the cosine integral.
  \item Give an example of an odd function and expand it in a sine series.
  \item Define a functional. Find the shortest smooth plane curve joining $P(x_1,y_1)$ and $Q(x_2,y_2)$.
\end{parts}
