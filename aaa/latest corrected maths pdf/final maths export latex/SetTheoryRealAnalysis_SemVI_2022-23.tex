\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Let $X = \mathbb{R}$ be the set of all real numbers. For $x, y \in X$, define $d(x, y) = |x^2 - y^2|$. Show that $d$ is not a metric on $X$. \pts{2}
  \item Define discrete metric space. Prove that in the discrete metric space $(X, d)$, every subset $G \subset X$ is an open set. \pts{2}
  \item Give an example of a pseudometric which is not a metric. \pts{2}
  \item Let $A$ be a subset of the metric space $(X, d)$. Prove that $\operatorname{diam}(A) = 0$ if and only if $A$ contains at most one point. Here $\operatorname{diam}(A)$ denotes the diameter of $A$. \pts{2}
  \item Let $(X, d)$ be a metric space. Prove that every finite set in $X$ is closed. \pts{2}
  \item Show that the set of all rational numbers $\mathbb{Q}$ is not open in the metric space $\mathbb{R}$ with the usual metric. \pts{2}
  \item By giving an example, show that $\overline{A \cap B} 
eq \overline{A} \cap \overline{B}$, where $A, B$ are two subsets of the metric space $(X, d)$ and $\overline{A}$ denotes the closure of $A$. \pts{2}
  \item In a metric space $(X, d)$, prove that a Cauchy sequence is bounded. \pts{2}
  \item Prove that $\mathbb{Z}$, the set of integers, is not dense in $\mathbb{R}$. \pts{2}
  \item Prove that $\operatorname{card}[0, 1) = \operatorname{card}(0, 1]$. \pts{2}
  \item Define partial order relation and give an example of a partial order relation. \pts{2}
\end{parts}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item Let $W$ be the space of all sequences in $\mathbb{R}$. Define
  \[
  d(x, y) = \sum_{n=1}^\infty \frac{1}{2^n} \frac{|x_n - y_n|}{1 + |x_n - y_n|}
  \]
  where $x = (x_n)$ and $y = (y_n)$ are in $W$. Prove that $(W, d)$ is a metric space. \pts{6}
  \item Let $(X, d)$ be a metric space and $Y \subset X$. Prove that a set $A$ is open in $(Y, d_Y)$ if and only if there exists a set $G$ open in $(X, d)$ such that $A = G \cap Y$. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 3.}

\begin{parts}
  \item Let $(X, d)$ be a metric space. Prove that every closed sphere in $X$ is a closed set. \pts{5}
  \item Let $(X, d)$ be a metric space and let the metric $d^*$ on $X$ be defined by
  \[
  d^*(x, y) = \frac{d(x, y)}{1 + d(x, y)}, \quad  \text{for } x, y \in X.
  \]
  Prove that $d^*$ is equivalent to $d$. \pts{7}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 4.} Let $A$ be a subset of a metric space $(X, d)$. Prove that: \pts{3 each}

\begin{parts}
  \item $X - A^\circ = \overline{X - A}$, where $A^\circ$ denotes the interior of $A$.
  \item $\operatorname{diam}(\overline{A}) = \operatorname{diam}(A)$, where $\operatorname{diam}(A)$ denotes the diameter of $A$.
  \item $A$ is closed if and only if $A$ contains all its limit points.
  \item Define boundary $\partial A$ of $A$. Prove that $\partial A = \overline{A} \setminus A^\circ$.
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 5.}

\begin{parts}
  \item Prove that the space $C[0, 1]$ of all continuous real-valued functions defined on $[0, 1]$ with the metric $d$ defined by
  \[
  d(f, g) = \sup \{ |f(x) - g(x)| : x \in [0, 1] \}
  \]
  is a complete metric space. \pts{6}
  \item Let $(X, d)$ and $(Y, \rho)$ be any two metric spaces. A function $f: X  o Y$ is continuous on $X$ if and only if for each open set $G \subset Y$, $f^{-1}(G)$ is an open subset of $X$. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 6.}

\begin{parts}
  \item Let $(X, d)$ and $(Y, \rho)$ be two metric spaces and $f: X  o Y$ be a bijective isometry. Then prove that $f$ is continuous. \pts{5}
  \item Let $(X, d)$ and $(Y, \rho)$ be two metric spaces and $f: X  o Y$ be a uniformly continuous function. Let $(x_n)_{n=1}^\infty$ be a Cauchy sequence in $X$. Prove that $(f(x_n))_{n=1}^\infty$ is a Cauchy sequence in $Y$. \pts{5}
  \item Define homeomorphism. Give an example of a homeomorphism. \pts{2}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 7.}

\begin{parts}
  \item Define 'countable set' and 'uncountable set'. Give one example of each. Prove that $(0, 1)$ is uncountable. \pts{6}
  \item Prove that a countable union of countable sets is countable. \pts{6}
\end{parts}
