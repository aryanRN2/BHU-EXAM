\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Define the order and degree of a differential equation.
  \item Obtain the relation in $a,b,c$ (constants) when $y=xe^{ax+bx^2+cx^3}$ is a solution of $y''+P(x)y'+Q(x)y=0$.
  \item Find an integrating factor for $y(axy+e^x)dx - e^x dy=0$.
  \item Solve $\dfrac{dy}{dx}+ay=e^{mx}$ where $a,m$ are constants.
  \item Obtain the differential equation of the family of plane curves $cy=x^2+c^2$.
  \item Let $y(x)$ be a solution of $\dfrac{dy}{dx}+3y=4$, $y(0)=3$. Find $\lim_{x o\infty}y(x)$.
  \item Find the value of $\dfrac{1}{(D+a)^2}\sin(ax)$ where $D=\dfrac{d}{dx}$.
  \item Find the condition for $y=e^{ax}$ to be a solution of $\dfrac{d^2y}{dx^2}+P(x)\dfrac{dy}{dx}+Q(x)y=0$.
  \item Write a 2nd order ODE with $x=1$ as a regular singular point and $x=0$ as an irregular singular point.
  \item Express $x^3+3x^2$ in terms of Legendre polynomials.
  \item Prove that $J_{-1/2}(x)=\sqrt{\dfrac{2}{\pi x}}\cos x$.
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item Solve the IVP: $\dfrac{dy}{dx}+y\cot x=\cos^2 x\,(1+y^2\sin^2 x)$, $y\!\left(\dfrac{\pi}{2}\right)=1$.\pts{6}
  \item Solve: $xyp^2+p(3x^2-2y^2)-6xy=0$ where $p=\dfrac{dy}{dx}$.\pts{6}
\end{parts}

\noindent   \textbf{Question 3.}

\begin{parts}
  \item Solve: $\dfrac{d^4y}{dx^4}-\dfrac{d^2y}{dx^2}=e^{-x}+2+x$.\pts{6}
  \item Solve: $\dfrac{d^2y}{dx^2}+a^2y=\sec(ax)$.\pts{6}
\end{parts}

\noindent   \textbf{Question 4.}

\begin{parts}
  \item If $f(D)=\sum_{r=0}^{n}a_r D^r$, establish:
  \[
    \frac{1}{f(D)}[e^{ax}f_2(x)]=e^{ax}\frac{1}{f_1(-a^2)+af_2(-a^2)}\left[\frac{x^n}{n!}\right]
  \]
  where $f(D)=f_1(D^2)+Df_2(D^2)$.\pts{6}
  \item Using variation of parameters, solve: $x^2\dfrac{d^2y}{dx^2}+x\dfrac{dy}{dx}-y=x^2\log x$, $x>0$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}
