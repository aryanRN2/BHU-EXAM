\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 4 marks each.)}

\begin{parts}
  \item Let $\gamma(s)$ be a unit speed plane curve given by $\gamma(s) = (x(s), y(s))$. Prove that the curvature of the curve $\gamma(s)$ is $|x'y'' - x''y'|$, where prime denotes derivative with respect to arc length $s$. \pts{3}
  \item Consider a closed curve in the plane with length equal to $6$ cm and bounding an area of $3$ cm$^2$. Does such a curve exist? Justify your answer. \pts{3}
  \item Give the definition of a simple curve. Prove that straight lines in $\mathbb{R}^n$ are simple curves.
  \item Give the definition of a simply connected domain. Also write down the statement of the Jordan curve theorem.
  \item Give the definition of an open hemisphere with pole $N$. Prove that the tangent spherical image of a regular closed curve $\alpha$ does not lie in any open hemisphere.
  \item Give the definition of a compact surface. Let $r$ be the minimum number such that $M \subset B_r(0)$. Prove that $M \cap S_r 
eq \emptyset$, where $S_r$ is the sphere of radius $r$ centered at the origin.
\end{parts}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item Give the definition of a closed curve. Show that the ellipse $\gamma(t) = (a\cos t, b\sin t, 0)$, $0 \le t \le 2\pi$ and $a 
eq b$, is a closed simple curve. \pts{6}
  \item State and prove the Rotation Index Theorem. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 3.}

\begin{parts}
  \item Let $A$ be the area of the region bounded by a simple closed regular plane curve $\gamma$ of length $L$. Prove that $L^2 \ge 4\pi A$. \pts{6}
  \item Prove that the four-vertex theorem is false if the hypothesis ``simple'' is omitted by considering the curve $\gamma(t) = (\cos t - 2\sin t\cos t, \sin t - 2\sin^2 t, 0)$, $0 \le t \le 2\pi$. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 4.}

\begin{parts}
  \item If $\alpha$ is an oval of constant width $w$, then prove that the length of $\alpha$ is $\pi w$. \pts{6}
  \item Prove that the ellipse $\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1$ with semi-axes $a > b > 0$ is an oval. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 5.}

\begin{parts}
  \item State Fenchel's theorem and verify it for the circle $\alpha(t) = (\cos t, \sin t, 0)$, $0 \le t \le 2\pi$. \pts{6}
  \item If $\alpha$ is a simple knotted regular closed curve in $\mathbb{R}^3$, then prove that the total curvature of $\alpha$ is at least $4\pi$. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 6.}

\begin{parts}
  \item Define the Weingarten map (or shape operator) $L$. Prove that the derivative of the unit normal vector is given by $L(X_i) = -D_{X_i} N = \sum h_i^j X_j$, where $h_i^j$ are the components of the shape operator. \pts{6}
  \item Give the definition of a triangulation of a surface $S$. State and prove the Gauss-Bonnet theorem for a compact surface. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}
