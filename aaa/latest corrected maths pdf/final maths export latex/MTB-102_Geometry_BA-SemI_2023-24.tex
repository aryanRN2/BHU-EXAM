\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item Find the latus rectum and the points on the conic $\ell/r = 3-2\cos \theta$ whose radius vector is $2$.
  \item Find the angle between the planes $x+y+z=1$ and $2x-3y+z=1$.
  \item Find the equation of the plane containing lines through the origin with direction cosines proportional to $(1,-2,2)$ and $(2,3,-1)$.
  \item Show that the line $\dfrac{x}{1}=\dfrac{y-5}{3}=\dfrac{z+8}{-4}$ lies on the plane $x+y+z=0$.
  \item Name the surface $x^2+y^2+z^2-x-2y-3z=0$ and state four points through which it passes.
  \item Write the equation of a sphere through the circle $x^2+y^2-x=0$, $z=0$.
  \item Write the equation of a cone whose vertex is $(0,0,0)$.
  \item Define right circular cylinder.
  \item Name the surface $\dfrac{x^2}{4}-\dfrac{y^2}{9}+\dfrac{z^2}{16}=1$ and tell which axis it does not intersect.
  \item Find the equation of the normal to $3x^2+4y^2+5z^2=64$ at $(1,2,3)$.
  \item The plane $2x-4y-z+3=0$ touches $x^2-2y^2=3z$. Find the point of contact.
\end{parts}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item Derive the equation of a conic in polar coordinates.\pts{6}
  \item Find the polar equation of the normal at any point $P$ (vectorial angle $\alpha$) of the conic $\ell/r=1-e\cos \theta$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 3.}

\begin{parts}
  \item A variable plane passes through $(a,b,c)$ and meets coordinate axes in $A,B,C$. Show the locus of the point common to the planes through $A,B,C$ parallel to coordinate planes is $\dfrac{a}{x}+\dfrac{b}{y}+\dfrac{c}{z}=1$.\pts{6}
  \item Show that the lines $\dfrac{x+4}{3}=\dfrac{y+6}{5}=\dfrac{z-3}{-2}$ and $\dfrac{x-3}{1}=\dfrac{y+5}{5}=\dfrac{z+1}{4}$ are coplanar. Find the plane containing them.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 4.}

\begin{parts}
  \item A sphere of constant radius $2k$ passes through the origin and meets the axes in $A,B,C$. Find the locus of the centroid of tetrahedron $OABC$.\pts{6}
  \item Find the sphere having the circle $x^2+y^2+z^2+10y-4z-8=0$, $x+y+z=3$ as its great circle.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 5.}

\begin{parts}
  \item Find the cone with vertex $(\alpha,\beta,\gamma)$ whose generators intersect $ax^2+2hxy+by^2+2gx+2fy+c=0$, $z=0$.\pts{6}
  \item A point $P$ moves so that the section of its enveloping cone of the ellipsoid $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}+\dfrac{z^2}{c^2}=1$ by $z=0$ is a circle. Show that $P$ lies on $\dfrac{x^2}{a^2-b^2}+\dfrac{z^2}{c^2}=1$.\pts{6}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\noindent   \textbf{Question 6.}

\begin{parts}

  \item Find the right circular cylinder of radius $2$ with axis $\dfrac{x-1}{2}=\dfrac{y-2}{2}=\dfrac{z-3}{1}$.\pts{6}
\end{parts}
