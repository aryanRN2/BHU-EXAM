\noindent\textbf{Question 1.} \textit{(Compulsory --- 2 marks each.)}
\begin{parts}
  \item Give a relation and a function from $A=\{1,2,3\}$ to $B=\{x,y,z\}$.
  \item Define the domain and range of a function. Find the domain for which $f(x)=2x^2-1$ and $g(x)=1-3x$ are equal.
  \item Evaluate $\displaystyle\lim_{x\to2}\frac{\sqrt{x+2}-2}{x-2}$.
  \item Show that $f(x)=7x-3$ is increasing on $\mathbb{R}$.
  \item Evaluate $\displaystyle\int \frac{1}{\sqrt{x}}\,dx$.
  \item Find $\displaystyle\int x\cos x\,dx$.
  \item Determine the order and degree of $(y'')^2+(y')^3+(y)^4+y^5=0$.
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 2.}
\begin{parts}
  \item Define the derivative of a real-valued function. Evaluate the derivative of $f(x)=\sin x$ using the definition.\pts{7}
  \item Find the equations of the tangent line and normal to the curve $y=x^2+3x+2$ at the point $(1,6)$. Explain its geometrical significance.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 3.}
\begin{parts}
  \item For any positive integer $n$, prove that $\displaystyle\lim_{x\to a}x^n=a^n$.\pts{7}
  \item What is the value of $\displaystyle\lim_{x\to0}\frac{\sin x}{x}$? Hence evaluate $\displaystyle\lim_{x\to0}\frac{\tan x}{x}$.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 4.}
\begin{parts}
  \item The volume of a cube is increasing at $9$ cm$^3$/s. How fast is the surface area increasing when the edge length is $10$ cm?\pts{7}
  \item Find the intervals on which $f(x)=\sin x+\cos x$, $0<x<2\pi$, is increasing or decreasing.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 5.}
\begin{parts}
  \item Find local maximum and minimum values of $f(x)=3x^4+4x^3-12x^2+12$.\pts{7}
  \item A manufacturer can sell $x$ items at price $\left(5-\dfrac{x}{100}\right)$ rupees each. The cost price of $x$ items is $\left(\dfrac{x}{5}+500\right)$ rupees. Find the number of items to sell for maximum profit.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 6.}
\begin{parts}
  \item Show that $\displaystyle\int \sec x\,dx = \log|\sec x+\tan x|+C$.\pts{7}
  \item Find $\displaystyle\int \frac{3x-2}{x^2+x-2}\,dx$.\pts{7}
\end{parts}
\medskip\rule{\linewidth}{0.2pt}

\medskip\noindent\textbf{Question 7.}
\begin{parts}
  \item In a bank, principal increases continuously at $5\%$ per year. In how many years does Rs.\,1000 double itself?\pts{7}
  \item Show that $(x-y)\dfrac{dy}{dx}=x+2y$ is homogeneous, and solve it.\pts{7}
\end{parts}
