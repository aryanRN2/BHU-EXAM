\noindent   \textbf{Question 1.}   \textit{(Compulsory --- 2 marks each.)}

\begin{parts}
  \item If $f(x) = |x|$, find $f(-1)$.
  \item Prove that $\lim_{x o 0} \frac{|x|}{x}$ does not exist.
  \item Examine the continuity of the function:
  \[ f(x) = 
\begin{cases} 1+x, & x \le 2 \\ 5-x, & x > 2 \end{cases} \]
  at $x = 2$.
  \item Find the value of $k$ if the function:
  \[ f(x) = 
\begin{cases} x-k, & x > 2 \\ kx^2, & x \le 2 \end{cases} \]
  is continuous at $x = 2$.
  \item Write the necessary condition for a function $f(x)$ to have a local maximum at $x = a$.
  \item State the order and the degree of the following differential equation:
  \[ x \frac{d^2y}{dx^2} + y \left(\frac{dy}{dx}\right)^2 + x = 0 \]
  \item Evaluate the definite integral:
  \[ \int_0^2 (t^2 + 1) \, dt \]
  \item Show that the function $f(x)=|x|$ is not differentiable at $x = 0$.
  \item Differentiate the following function with respect to $x$:
  \[ y = (ax)^n + b^m \]
  \item Find the points on the curve $y=x^3-3x^2+3x$, where the gradient (slope) is zero.
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\noindent   \textbf{Question 2.}

\begin{parts}
  \item Evaluate the limit:
  \[ \lim_{x o -1} \frac{x^2 - 1}{x^2 + 3x + 2} \] \pts{6}
  \item A function $f(x)$ is defined as:
  \[ f(x) = 
\begin{cases} 1, &  \text{when } x 
eq 0 \\ 0, &  \text{when } x = 0 \end{cases} \]
  Does the limit of $f(x)$ as $x  o 0$ exist? Justify your answer. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 3.}

\begin{parts}
  \item Show that the function $f(x)$ defined as:
  \[ f(x) = 
\begin{cases} \frac{x^2 - 1}{x - 1}, & x 
eq 1 \\ 2, & x = 1 \end{cases} \]
  is continuous at $x = 1$. \pts{6}
  \item Let:
  \[ f(x) = 
\begin{cases} x^2, &  \text{if } x \le 1 \\ 2, &  \text{if } x > 1 \end{cases} \]
  Show that $f(x)$ is discontinuous at $x = 1$. \pts{6}
\end{parts}

\medskip\rule{\linewidth}{0.2pt}

\medskip


\noindent   \textbf{Question 4.}

\begin{parts}
  \item Find the derivative with respect to $x$ of the function:
  \[ y = \frac{x^3 + x^2}{x^2 + 1} \] \pts{6}
  \item If $y = ax + \frac{b}{x}$, then prove that:
  \[ x^2 \frac{d^2y}{dx^2} + x \frac{dy}{dx} - y = 0 \] \pts{6}
\end{parts}
