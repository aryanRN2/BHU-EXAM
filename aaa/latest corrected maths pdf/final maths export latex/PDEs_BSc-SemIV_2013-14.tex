\noindent\textbf{Question 1.}
\begin{parts}
  \item Find the PDE by eliminating the arbitrary constants $a$ and $b$ from $ax^2 + by^2 + z^2 = 1$. \pts{2}
  \item Eliminate the arbitrary function $f$ from $z = f(x - y)$. \pts{2}
  \item Find the general integral of $xp + yq = z$. \pts{2}
  \item Find the complete integral of $pq = xy$. \pts{2}
  \item Find the complete integral of $pq = p + q$. \pts{2}
  \item Solve $2r + 5s + 2t = 0$. \pts{2}
  \item Solve $(D + D' - 1)(D + 2D' - 3)z = 0$. \pts{2}
  \item Solve $z = px + qy + \ln(pq)$. \pts{2}
  \item Solve $\dfrac{\partial^2 z}{\partial y^2} = \sin(xy)$ (i.e.\ $t = \sin xy$). \pts{2}
  \item Solve $x\,r + 2p = 0$. \pts{2}
  \item Solve $\dfrac{\partial u}{\partial x} = 2\dfrac{\partial u}{\partial y}$ given $u(0, y) = 8e^{-3y}$. \pts{2}
\end{parts}
